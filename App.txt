// ─── CROSS-PLATFORM HELPERS ────────────────────────────────────────────────
const showAlert = (title, message) => {
  if (Platform.OS === 'web') {
    window.alert(message ? `${title}\n\n${message}` : title);
  } else {
    Alert.alert(title, message);
  }
};

const showConfirm = (title, message, onConfirm) => {
  if (Platform.OS === 'web') {
    const result = window.confirm(message ? `${title}\n\n${message}` : title);
    if (result) onConfirm();
  } else {
    Alert.alert(title, message, [
      { text: 'Cancel', style: 'cancel' },
      { text: 'Confirm', onPress: onConfirm },
    ]);
  }
};

  async signUp(email, password, fullName, businessName) {
    try {

        body: JSON.stringify({

          data: { full_name: fullName, business_name: businessName },

      });
      const data = await res.json();

  async getTodaySummary(userId) {
    try {
      const today = new Date().toISOString().slice(0, 10);
      // Week start (Sunday)
      const now = new Date();
      // Week starts Monday (business week). getDay(): Sun=0, Mon=1 ... Sat=6.
      const day = now.getDay();
      const daysSinceMonday = (day === 0) ? 6 : (day - 1);
      const weekStart = new Date(now);
      weekStart.setDate(now.getDate() - daysSinceMonday);
      const weekStartStr = weekStart.toISOString().slice(0, 10);

      const [apptRes, timeRes] = await Promise.all([
        fetch(`${SUPABASE_URL}/rest/v1/bp_appointments?user_id=eq.${userId}&archived=eq.false&appointment_date=eq.${today}&order=start_time.asc`, { headers: authHeaders() }),
        fetch(`${SUPABASE_URL}/rest/v1/bp_time_entries?user_id=eq.${userId}&archived=eq.false&entry_date=gte.${weekStartStr}&select=hours`, { headers: authHeaders() }),
      ]);
      const appts = await apptRes.json();
      const times = await timeRes.json();
      const weekHours = (Array.isArray(times) ? times : []).reduce((sum, t) => sum + (parseFloat(t.hours) || 0), 0);
      return {
        todayAppointments: Array.isArray(appts) ? appts : [],
        weekHours: weekHours,
      };
    } catch (e) { return { todayAppointments: [], weekHours: 0 }; }
  },

// ─── FREE TIER LIMITS ─────────────────────────────────────────────────────
const FREE_LIMITS = {
  clients: 3,
  proposals: 5,
  invoices: 5,
  documents: 10,
  contracts: 2,   // per calendar month
  ai: 20,         // BizHQ AI questions per calendar month
};
const MONTHLY_LIMITS = ['contracts', 'ai'];

const checkFreeLimit = async (userId, subscriptionStatus, type) => {
  if (_planStatus === 'pro' || subscriptionStatus === 'pro') return { allowed: true };
  const limit = FREE_LIMITS[type];
  if (!limit) return { allowed: true };
  try {
    const tableMap = { clients: 'bp_clients', proposals: 'bp_proposals', invoices: 'bp_invoices', documents: 'bp_documents', contracts: 'bp_contracts', ai: 'bp_ai_commands' };
    const table = tableMap[type];
    const monthly = MONTHLY_LIMITS.includes(type);
    let dateFilter = '';
    if (monthly) {
      const now = new Date();
      const monthStart = new Date(now.getFullYear(), now.getMonth(), 1).toISOString();
      dateFilter = `&created_at=gte.${encodeURIComponent(monthStart)}`;
    }
    const res = await fetch(
      `${SUPABASE_URL}/rest/v1/${table}?user_id=eq.${userId}${dateFilter}&select=id`,
      { headers: authHeaders() }
    );
    const data = await res.json();
    const count = Array.isArray(data) ? data.length : 0;
    if (count >= limit) {
      return {
        allowed: false,
        message: monthly
          ? `The Free plan includes ${limit} ${type === 'ai' ? 'BizHQ AI questions' : type} per month. Upgrade to Pro in Settings for unlimited access.`
          : `You've reached the free plan limit of ${limit} ${type}. Upgrade to Pro in Settings for unlimited access.`,
      };
    }
    return { allowed: true };
  } catch (e) { return { allowed: true }; }
};

function LoginScreen({ onLogin, onGoSignUp }) {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);
  const [showPassword, setShowPassword] = useState(false);
  const [resetLoading, setResetLoading] = useState(false);
  const [rememberMe, setRememberMe] = useState(false);

  useEffect(() => {
    AsyncStorage.getItem('bp_savedEmail').then(saved => {
      if (saved) { setEmail(saved); setRememberMe(true); }
    }).catch(() => {});
  }, []);

  const handleLogin = async () => {
    if (!email.trim()) { showAlert('Required', 'Please enter your email.'); return; }
    if (!password.trim()) { showAlert('Required', 'Please enter your password.'); return; }
    setLoading(true);
    try {
      const result = await supabase.signIn(email.trim(), password);
      if (result.error) {
        showAlert('Login Failed', result.error);
      } else if (result.user) {
        const isDeactivated = await supabase.checkDeletion(result.user.id);
        if (isDeactivated) {
          showAlert('Account Deactivated', 'This account has been deactivated. Contact support@invoicespro.org to recover it.');
          setLoading(false);
          return;
        }
        try {
          if (rememberMe) {
            await AsyncStorage.setItem('bp_savedEmail', email.trim());
            await AsyncStorage.setItem('bp_savedSession', JSON.stringify({
              user: result.user,
              refresh_token: result.session.refresh_token,
            }));
          } else {
            await AsyncStorage.removeItem('bp_savedEmail');
            await AsyncStorage.removeItem('bp_savedSession');
          }
        } catch (e) {}
        onLogin(result.user, result.session);
      } else {
        showAlert('Login Failed', 'Could not sign in. Please check your credentials.');
      }
    } catch (e) {
      showAlert('Login Failed', e.message || 'Something went wrong.');
    }
    setLoading(false);
  };

  const handleForgotPassword = async () => {
    if (!email.trim()) {
      showAlert('Email Required', 'Enter your email above, then tap "Forgot password?" to receive a reset link.');
      return;
    }
    setResetLoading(true);
    const result = await supabase.resetPassword(email.trim());
    setResetLoading(false);
    if (result.error) {
      showAlert('Reset Failed', result.error);
    } else {
      showAlert('Check Your Email', `A password reset link has been sent to ${email.trim()}.`);
    }
  };

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
        <ScrollView
          contentContainerStyle={{ flexGrow: 1, justifyContent: 'center', paddingHorizontal: H, paddingBottom: 40 }}
          keyboardShouldPersistTaps="handled"
        >
          <View style={{ alignItems: 'center', marginBottom: 48 }}>
            <View style={{ flexDirection: 'row', alignItems: 'baseline', marginBottom: 8 }}>
              <Text style={[st.appName, { color: C.textDark }]}>Biz</Text>
              <Text style={[st.appNameAccent, { color: C.primary }]}>HQ</Text>
            </View>
            <Text style={{ fontSize: 14, color: C.textMute }}>Your business headquarters.</Text>
          </View>

          <View style={[st.card, { padding: 20, marginBottom: 16 }]}>
            <Text style={st.fieldLabel}>Email</Text>
            <TextInput
              style={st.fieldInput}
              value={email}
              onChangeText={setEmail}
              placeholder="your@email.com"
              placeholderTextColor={C.textMute}
              keyboardType="email-address"
              autoCapitalize="none"
              autoComplete="email"
              textContentType="emailAddress"
            />
            <Text style={[st.fieldLabel, { marginTop: 14 }]}>Password</Text>
            <View style={{ flexDirection: 'row', alignItems: 'center' }}>
              <TextInput
                style={[st.fieldInput, { flex: 1 }]}
                value={password}
                onChangeText={setPassword}
                placeholder="Your password"
                placeholderTextColor={C.textMute}
                secureTextEntry={!showPassword}
                autoComplete="password"
                textContentType="password"
              />
              <TouchableOpacity onPress={() => setShowPassword(!showPassword)} style={{ padding: 10, marginLeft: 4 }}>
                <Ionicons name={showPassword ? 'eye-off-outline' : 'eye-outline'} size={20} color={C.textMute} />
              </TouchableOpacity>
            </View>
            <TouchableOpacity onPress={handleForgotPassword} disabled={resetLoading} style={{ alignSelf: 'flex-end', marginTop: 10 }}>
              <Text style={{ fontSize: 13, color: '#FFFFFF', fontWeight: '600' }}>
                {resetLoading ? 'Sending...' : 'Forgot password?'}
              </Text>
            </TouchableOpacity>
            <TouchableOpacity onPress={() => setRememberMe(!rememberMe)} style={{ flexDirection: 'row', alignItems: 'center', marginTop: 14 }}>
              <View style={{
                width: 20, height: 20, borderRadius: 5, borderWidth: 1.5,
                borderColor: rememberMe ? C.primary : C.border,
                backgroundColor: rememberMe ? C.primary : C.white,
                alignItems: 'center', justifyContent: 'center', marginRight: 10,
              }}>
                {rememberMe && <Ionicons name="checkmark" size={13} color={C.white} />}
              </View>
              <Text style={{ fontSize: 13, color: C.textDark }}>Remember me</Text>
            </TouchableOpacity>
          </View>

          <TouchableOpacity
            style={[st.primaryBtn, { opacity: loading ? 0.7 : 1 }]}
            onPress={handleLogin}
            disabled={loading}
            activeOpacity={0.8}
          >
            {loading
              ? <ActivityIndicator color={C.white} size="small" />
              : <Text style={st.primaryBtnText}>Sign In</Text>
            }
          </TouchableOpacity>

          <TouchableOpacity onPress={onGoSignUp} style={{ marginTop: 24, alignItems: 'center' }}>
            <Text style={{ fontSize: 14, color: C.textMute }}>
              Don't have an account?{' '}
              <Text style={{ color: C.primary, fontWeight: '700' }}>Sign Up</Text>
            </Text>
          </TouchableOpacity>
        </ScrollView>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

// ─── SIGN UP SCREEN ────────────────────────────────────────────────────────
function SignUpScreen({ onSignUp, onGoLogin }) {
  const [fullName, setFullName] = useState('');
  const [businessName, setBusinessName] = useState('');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);
  const [showPassword, setShowPassword] = useState(false);

  const handleSignUp = async () => {
    if (!fullName.trim()) { showAlert('Required', 'Please enter your full name.'); return; }
    if (!businessName.trim()) { showAlert('Required', 'Please enter your business name.'); return; }
    if (!email.trim()) { showAlert('Required', 'Please enter your email.'); return; }
    if (!password.trim() || password.length < 6) { showAlert('Required', 'Password must be at least 6 characters.'); return; }
    setLoading(true);
    const result = await supabase.signUp(email.trim(), password, fullName.trim(), businessName.trim());
    if (result.error) {
      showAlert('Sign Up Failed', result.error);
      setLoading(false);
      return;
    }
    if (result.user) {
      await supabase.createProfile(result.user.id, fullName.trim(), businessName.trim());
    }

// ─── ONBOARDING SCREEN ─────────────────────────────────────────────────────
function OnboardingScreen({ onFinish, user }) {
  const [page, setPage] = useState(0);
  const slides = [
    { icon: 'people-outline', title: 'Your clients, organized', desc: 'Keep every client, proposal, contract, and invoice in one place. No more digging through emails.' },
    { icon: 'document-text-outline', title: 'Proposals that get signed', desc: 'Send a professional proposal in minutes. One tap to convert it to a contract. One tap to send for signature.' },
    { icon: 'card-outline', title: 'Get paid faster', desc: 'Stripe-powered payment links on every invoice. Clients pay from their phone. You get notified the moment it lands.' },
    { icon: 'chatbubble-ellipses-outline', title: 'Just tell BizHQ', desc: '"Send Sandra a reminder." "What does James still owe?" Your AI assistant handles the back office so you can focus on the work.' },
  ];
  // The setup step is one past the last slide.
  const setupPage = slides.length;
  const isSetup = page === setupPage;
  const isLastSlide = page === slides.length - 1;
  const slide = slides[page] || slides[0];

  const [bizName, setBizName] = useState('');
  const [logoUri, setLogoUri] = useState(null);
  const [saving, setSaving] = useState(false);

  const handlePickLogo = () => {
    if (Platform.OS === 'web') {
      showAlert('Not supported', 'Logo upload is not available on web. Use the phone app.');
      return;
    }
    Alert.alert(
      'Upload Logo',
      'Choose where to find your logo',
      [
        {
          text: 'Photo Library',
          onPress: async () => {
            try {
              const perm = await ImagePicker.requestMediaLibraryPermissionsAsync();
              if (!perm.granted) { showAlert('Permission needed', 'Please allow photo access.'); return; }
              const mediaType = ImagePicker.MediaTypeOptions?.Images ?? ImagePicker.MediaType?.images ?? 'Images';
              const result = await ImagePicker.launchImageLibraryAsync({ mediaTypes: mediaType, allowsEditing: true, aspect: [1, 1], quality: 0.7, base64: true });
              if (!result.canceled && result.assets?.[0]) setLogoUri(result.assets[0]);
            } catch (e) { showAlert('Error', 'Could not open photo library.'); }
          },
        },
        {
          text: 'Files',
          onPress: async () => {
            try {
              const result = await DocumentPicker.getDocumentAsync({ type: ['image/png', 'image/jpeg', 'image/jpg'], copyToCacheDirectory: true, base64: true });
              if (!result.canceled && result.assets?.[0]) {
                setLogoUri({ uri: result.assets[0].uri, base64: result.assets[0].base64, mimeType: result.assets[0].mimeType || 'image/png' });
              }
            } catch (e) { showAlert('Error', 'Could not open files.'); }
          },
        },
        { text: 'Cancel', style: 'cancel' },
      ]
    );
  };

  const finishOnboarding = async () => {
    try { await AsyncStorage.setItem('bp_onboardingComplete', 'true'); } catch (e) {}
    onFinish();
  };

  const handleSaveSetup = async () => {
    setSaving(true);
    try {
      const updates = {};
      if (bizName.trim()) updates.business_name = bizName.trim();
      if (logoUri?.uri && user) {
        const logoUrl = await supabase.uploadLogo(user.id, logoUri.uri, logoUri.mimeType || 'image/jpeg');
        if (logoUrl) updates.logo_url = logoUrl;
      }
      if (Object.keys(updates).length > 0 && user) {
        await supabase.updateProfile(user.id, updates);
      }
    } catch (e) {}
    setSaving(false);
    finishOnboarding();
  };

  // ── SETUP STEP ──
  if (isSetup) {
    return (
      <SafeAreaView style={{ flex: 1, backgroundColor: C.background }}>
        <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
        <ScrollView contentContainerStyle={{ flexGrow: 1, justifyContent: 'center', paddingHorizontal: 36, paddingVertical: 40 }} keyboardShouldPersistTaps="handled">
          <Text style={{ fontSize: 24, fontWeight: '700', color: C.textDark, textAlign: 'center', marginBottom: 10, letterSpacing: -0.3 }}>
            Set up your business
          </Text>
          <Text style={{ fontSize: 15, color: C.textMute, textAlign: 'center', lineHeight: 22, marginBottom: 36 }}>
            Add your logo and business name. They appear on your client portal and documents. You can change these any time in Settings.
          </Text>

          <View style={{ alignItems: 'center', marginBottom: 28 }}>
            <TouchableOpacity onPress={handlePickLogo} activeOpacity={0.8}>
              {logoUri ? (
                <Image source={{ uri: logoUri.uri }} style={{ width: 100, height: 100, borderRadius: 24, borderWidth: 2, borderColor: C.primaryLight }} />
              ) : (
                <View style={{ width: 100, height: 100, borderRadius: 24, backgroundColor: C.cardAlt, borderWidth: 1.5, borderColor: C.primaryLight, alignItems: 'center', justifyContent: 'center' }}>
                  <Ionicons name="image-outline" size={34} color={C.primary} />
                </View>
              )}
            </TouchableOpacity>
            <Text style={{ fontSize: 13, color: C.textMute, marginTop: 12 }}>{logoUri ? 'Tap to change logo' : 'Tap to add your logo'}</Text>
          </View>

          <Text style={st.fieldLabel}>Business name</Text>
          <TextInput style={st.fieldInput} value={bizName} onChangeText={setBizName} placeholder="Your business name" placeholderTextColor={C.textMute} />

          <TouchableOpacity onPress={handleSaveSetup} style={[st.primaryBtn, { width: '100%', marginTop: 28 }]} activeOpacity={0.8} disabled={saving}>
            <Text style={st.primaryBtnText}>{saving ? 'Saving...' : 'Finish Setup'}</Text>
          </TouchableOpacity>
          <TouchableOpacity onPress={finishOnboarding} style={{ marginTop: 18, alignItems: 'center' }}>
            <Text style={{ fontSize: 14, color: C.textMute }}>Skip for now</Text>
          </TouchableOpacity>
        </ScrollView>
      </SafeAreaView>
    );
  }

  // ── INTRO SLIDES ──
  return (
    <SafeAreaView style={{ flex: 1, backgroundColor: C.background }}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      <View style={{ flex: 1, justifyContent: 'center', alignItems: 'center', paddingHorizontal: 36 }}>
        <View style={{
          width: 88, height: 88, borderRadius: 24, backgroundColor: C.cardAlt,
          borderWidth: 1.5, borderColor: C.primaryLight,
          alignItems: 'center', justifyContent: 'center', marginBottom: 36,
        }}>
          <Ionicons name={slide.icon} size={40} color={C.primary} />
        </View>
        <Text style={{ fontSize: 24, fontWeight: '700', color: C.textDark, textAlign: 'center', marginBottom: 14, letterSpacing: -0.3 }}>
          {slide.title}
        </Text>
        <Text style={{ fontSize: 15, color: C.textMute, textAlign: 'center', lineHeight: 24, marginBottom: 52 }}>
          {slide.desc}
        </Text>
        <View style={{ flexDirection: 'row', marginBottom: 36 }}>
          {slides.map((_, i) => (
            <View key={i} style={{
              width: i === page ? 28 : 8, height: 8, borderRadius: 4,
              backgroundColor: i === page ? C.primary : C.border,
              marginHorizontal: 4,
            }} />
          ))}
        </View>
        <TouchableOpacity
          onPress={() => setPage(page + 1)}
          style={[st.primaryBtn, { width: '100%' }]}
          activeOpacity={0.8}
        >
          <Text style={st.primaryBtnText}>{isLastSlide ? 'Next: Set Up' : 'Next'}</Text>
        </TouchableOpacity>
        <TouchableOpacity onPress={finishOnboarding} style={{ marginTop: 18 }}>
          <Text style={{ fontSize: 14, color: C.textMute }}>Skip</Text>
        </TouchableOpacity>
      </View>
    </SafeAreaView>
  );
}

          )}

      )}

// ─── SETTINGS SCREEN ───────────────────────────────────────────────────────
function ArchivedScreen({ user, onBack, onNavigate }) {
  const [data, setData] = useState({ invoices: [], proposals: [], contracts: [] });
  const [tab, setTab] = useState('invoices');
  const [loading, setLoading] = useState(true);

  const load = async () => {
    setLoading(true);
    const d = await supabase.getArchivedItems(user.id);
    setData(d);
    setLoading(false);
  };
  useEffect(() => { load(); }, []);

  const fmtMoney = (n) => (n == null ? '' : new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(n));

  const handleRestore = (table, id, label) => {
    showConfirm('Restore ' + label, 'This moves it back to your active list.', async () => {
      const ok = await supabase.restoreItem(table, id);
      if (ok) { showAlert('Restored', label + ' is back in your active list.'); load(); }
      else { showAlert('Error', 'Could not restore. Try again.'); }
    });
  };

  const tabs = [
    { key: 'invoices', label: 'Invoices (' + data.invoices.length + ')' },
    { key: 'proposals', label: 'Proposals (' + data.proposals.length + ')' },
    { key: 'contracts', label: 'Contracts (' + data.contracts.length + ')' },
  ];

  const renderItems = () => {
    const list = data[tab];
    if (!list || list.length === 0) {
      return (
        <View style={{ alignItems: 'center', paddingTop: 50 }}>
          <Ionicons name="archive-outline" size={40} color={C.border} style={{ marginBottom: 12 }} />
          <Text style={{ color: C.textMute, fontSize: 14 }}>Nothing archived here</Text>
        </View>
      );
    }
    const tableName = tab === 'invoices' ? 'bp_invoices' : tab === 'proposals' ? 'bp_proposals' : 'bp_contracts';
    const singular = tab === 'invoices' ? 'Invoice' : tab === 'proposals' ? 'Proposal' : 'Contract';
    const amountField = tab === 'proposals' || tab === 'contracts' ? 'amount' : 'total_due';
    return list.map((item) => (
      <View key={item.id} style={[st.card, { flexDirection: 'row', alignItems: 'center', padding: 16, marginBottom: 10 }]}>
        <View style={{ flex: 1 }}>
          <Text style={{ fontSize: 15, fontWeight: '700', color: C.textDark, marginBottom: 2 }} numberOfLines={1}>{item.title || singular}</Text>
          <Text style={{ fontSize: 13, color: C.textMute }}>{fmtMoney(item[amountField])}{item.status ? ' · ' + item.status : ''}</Text>
        </View>
        <TouchableOpacity
          style={[st.outlineBtn, { borderColor: C.primary, paddingVertical: 8, paddingHorizontal: 14 }]}
          onPress={() => handleRestore(tableName, item.id, singular)}
          activeOpacity={0.8}
        >
          <Text style={[st.outlineBtnText, { color: C.primary, fontSize: 13 }]}>Restore</Text>
        </TouchableOpacity>
      </View>
    ));
  };

  return (
    <SafeAreaView style={st.safe}>
      <ScreenHeader title="Archived Items" onBack={onBack} />
      <View style={{ paddingHorizontal: H, paddingBottom: 8 }}>
        <ScrollView horizontal showsHorizontalScrollIndicator={false}>
          {tabs.map((t) => (
            <TouchableOpacity
              key={t.key}
              onPress={() => setTab(t.key)}
              style={{ paddingVertical: 8, paddingHorizontal: 16, marginRight: 8, borderRadius: 20, backgroundColor: tab === t.key ? C.primary : C.cardAlt }}
              activeOpacity={0.7}
            >
              <Text style={{ fontSize: 13, fontWeight: '600', color: tab === t.key ? C.white : C.textMid }}>{t.label}</Text>
            </TouchableOpacity>
          ))}
        </ScrollView>
      </View>
      <ScrollView contentContainerStyle={{ padding: H, paddingTop: 12, paddingBottom: 40 }}>
        {loading ? <ActivityIndicator color={C.primary} style={{ marginTop: 40 }} /> : renderItems()}
      </ScrollView>
    </SafeAreaView>
  );
}

// ─── EDIT CLIENT SCREEN ───────────────────────────────────────────────────
function EditClientScreen({ user, clientId, onBack, onSaved }) {
  const [name, setName] = useState('');
  const [email, setEmail] = useState('');
  const [phone, setPhone] = useState('');
  const [company, setCompany] = useState('');
  const [notes, setNotes] = useState('');
  const [smsConsent, setSmsConsent] = useState(false);
  const [status, setStatus] = useState('active');
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    supabase.getClient(clientId).then(client => {
      if (client) {
        setName(client.name || '');
        setEmail(client.email || '');
        const digits = (client.phone || '').replace(/\D/g, '').slice(-10);
        if (digits.length === 10) setPhone(`(${digits.slice(0,3)}) ${digits.slice(3,6)}-${digits.slice(6)}`);
        else setPhone(client.phone || '');
        setCompany(client.company || '');
        setNotes(client.notes || '');
        setSmsConsent(client.sms_consent || false);
        setStatus(client.status || 'active');
      }
      setLoading(false);
    });
  }, [clientId]);

  const formatPhone = (val) => {
    const digits = val.replace(/\D/g, '');
    if (digits.length <= 3) return digits;
    if (digits.length <= 6) return `(${digits.slice(0,3)}) ${digits.slice(3)}`;
    return `(${digits.slice(0,3)}) ${digits.slice(3,6)}-${digits.slice(6,10)}`;
  };

  const handleSave = async () => {
    if (!name.trim()) { showAlert('Required', 'Please enter the client name.'); return; }
    setSaving(true);
    const digits = phone.replace(/\D/g, '');
    const formattedPhone = digits.length >= 10 ? `+1${digits.slice(-10)}` : null;
    const updates = {
      name: name.trim(),
      email: email.trim() || null,
      phone: formattedPhone,
      company: company.trim() || null,
      notes: notes.trim() || null,
      sms_consent: smsConsent,
      status,
    };
    const ok = await supabase.updateClient(clientId, updates);
    setSaving(false);
    if (ok) {
      onSaved(clientId);
    } else {
      showAlert('Error', 'Could not save changes. Please try again.');
    }
  };

  if (loading) return (
    <SafeAreaView style={st.safe}>
      <ScreenHeader title="Edit Client" onBack={onBack} />
      <ActivityIndicator color={C.primary} style={{ marginTop: 60 }} />
    </SafeAreaView>
  );

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      <ScreenHeader
        title="Edit Client"
        onBack={onBack}
        rightAction={
          <TouchableOpacity onPress={handleSave} disabled={saving} style={{ paddingRight: 4 }}>
            {saving ? <ActivityIndicator color={C.primary} size="small" /> : <Text style={{ fontSize: 15, fontWeight: '700', color: C.primary }}>Save</Text>}
          </TouchableOpacity>
        }
      />
      <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
        <ScrollView contentContainerStyle={{ padding: H, paddingBottom: 60 }} keyboardShouldPersistTaps="handled" showsVerticalScrollIndicator={false}>

          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={st.sectionLabel}>CONTACT INFO</Text>
            <Text style={st.fieldLabel}>Full Name *</Text>
            <TextInput style={st.fieldInput} value={name} onChangeText={setName} placeholder="Full name" placeholderTextColor={C.textMute} autoCapitalize="words" />
            <Text style={[st.fieldLabel, { marginTop: 14 }]}>Email</Text>
            <TextInput style={st.fieldInput} value={email} onChangeText={setEmail} placeholder="their@email.com" placeholderTextColor={C.textMute} keyboardType="email-address" autoCapitalize="none" />
            <Text style={[st.fieldLabel, { marginTop: 14 }]}>Phone</Text>
            <TextInput style={st.fieldInput} value={phone} onChangeText={val => setPhone(formatPhone(val))} placeholder="(555) 000-0000" placeholderTextColor={C.textMute} keyboardType="phone-pad" maxLength={14} />
            {phone.replace(/\D/g,'').length >= 10 && (
              <TouchableOpacity onPress={() => setSmsConsent(!smsConsent)} style={{ flexDirection: 'row', alignItems: 'center', marginTop: 12 }}>
                <View style={{ width: 20, height: 20, borderRadius: 5, borderWidth: 1.5, borderColor: smsConsent ? C.primary : C.border, backgroundColor: smsConsent ? C.primary : C.white, alignItems: 'center', justifyContent: 'center', marginRight: 10 }}>
                  {smsConsent && <Ionicons name="checkmark" size={13} color={C.white} />}
                </View>
                <Text style={{ fontSize: 13, color: C.textDark, flex: 1 }}>Client agrees to receive SMS messages</Text>
              </TouchableOpacity>
            )}
          </View>

          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={st.sectionLabel}>BUSINESS</Text>
            <Text style={st.fieldLabel}>Company or Business Name</Text>
            <TextInput style={st.fieldInput} value={company} onChangeText={setCompany} placeholder="Optional" placeholderTextColor={C.textMute} autoCapitalize="words" />
          </View>

          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={st.sectionLabel}>NOTES</Text>
            <TextInput style={[st.fieldInput, { height: 100, textAlignVertical: 'top', paddingTop: 12 }]} value={notes} onChangeText={setNotes} placeholder="Private notes about this client..." placeholderTextColor={C.textMute} multiline maxLength={500} />
            <Text style={{ fontSize: 11, color: C.textMute, marginTop: 6, textAlign: 'right' }}>{notes.length}/500</Text>
          </View>

          <View style={[st.card, { padding: 16, marginBottom: 24 }]}>
            <Text style={st.sectionLabel}>STATUS</Text>
            {['active', 'inactive', 'archived'].map(s => (
              <TouchableOpacity key={s} onPress={() => setStatus(s)} style={{ flexDirection: 'row', alignItems: 'center', paddingVertical: 10 }}>
                <View style={{ width: 20, height: 20, borderRadius: 10, borderWidth: 1.5, borderColor: status === s ? C.primary : C.border, backgroundColor: status === s ? C.primary : C.white, alignItems: 'center', justifyContent: 'center', marginRight: 12 }}>
                  {status === s && <View style={{ width: 8, height: 8, borderRadius: 4, backgroundColor: C.white }} />}

                >
                  <Text style={[{ fontSize: 12, fontWeight: '600', color: C.textMute }, activeTab === tab.key && { color: C.white }]}>
                    {tab.label}
                  </Text>
                </TouchableOpacity>
              ))}
            </View>
          </ScrollView>
          {renderTabContent()}
        </View>
      </ScrollView>
    </SafeAreaView>
  );
}







// ─── DOCUMENTS SCREEN ─────────────────────────────────────────────────────
function DocumentsScreen({ user, onBack, onNavigate, screenParam }) {
  const [documents, setDocuments] = useState([]);
  const [clients, setClients] = useState([]);
  const [loading, setLoading] = useState(true);
  const [uploading, setUploading] = useState(false);
  const [filter, setFilter] = useState('all');
  const [showTagPicker, setShowTagPicker] = useState(null);
  const [docSearch, setDocSearch] = useState('');
  const [tagSearch, setTagSearch] = useState('');

  // Pre-selected client from client profile nav
  const preselectedClientId = screenParam?.preselectedClientId || null;
  const preselectedClientName = screenParam?.preselectedClientName || null;

  const load = async () => {
    setLoading(true);
    const [docs, cls] = await Promise.all([
      supabase.getDocuments(user.id),
      supabase.getClients(user.id),
    ]);
    setDocuments(docs);
    setClients(cls);
    setLoading(false);
  };

  useEffect(() => { load(); }, []);

  const handleUpload = () => {
    if (Platform.OS === 'web') { showAlert('Not supported', 'Use the phone app to upload documents.'); return; }
    Alert.alert(
      'Upload Document',
      'Choose where to find your document',
      [
        {
          text: 'Photo Library',
          onPress: async () => {
            try {
              const perm = await ImagePicker.requestMediaLibraryPermissionsAsync();
              if (!perm.granted) { showAlert('Permission needed', 'Please allow photo access.'); return; }
              const mediaType = ImagePicker.MediaTypeOptions?.Images ?? ImagePicker.MediaType?.images ?? 'Images';
              const result = await ImagePicker.launchImageLibraryAsync({ mediaTypes: mediaType, quality: 0.8 });
              if (!result.canceled && result.assets?.[0]) {
                const asset = result.assets[0];
                const fileName = asset.uri.split('/').pop() || `image_${Date.now()}.jpg`;
                await uploadFile(asset.uri, fileName, asset.mimeType || 'image/jpeg');
              }
            } catch (e) { showAlert('Error', 'Could not open photo library.'); }
          },
        },
        {
          text: 'Files',
          onPress: async () => {
            try {
              const result = await DocumentPicker.getDocumentAsync({
                type: '*/*',
                copyToCacheDirectory: true,
              });
              if (!result.canceled && result.assets?.[0]) {
                const asset = result.assets[0];
                await uploadFile(asset.uri, asset.name, asset.mimeType || 'application/octet-stream');
              }
            } catch (e) { showAlert('Error', 'Could not open files.'); }
          },
        },
        { text: 'Cancel', style: 'cancel' },
      ]
    );
  };

  const uploadFile = async (uri, name, mimeType) => {
    setUploading(true);
    const clientId = preselectedClientId || null;
    const result = await supabase.uploadDocument(user.id, uri, name, mimeType, clientId);
    setUploading(false);
    if (result) {
      showAlert('Uploaded', `"${name}" has been saved to your documents.`);
      load();
    } else {
      showAlert('Error', 'Could not upload document. Please try again.');
    }
  };

  const handleTag = async (docId, clientId) => {
    const ok = await supabase.tagDocumentToClient(docId, clientId);
    if (ok) { showAlert('Tagged', 'Document linked to client.'); load(); }
    setShowTagPicker(null);
  };

  const handleDelete = (doc) => {
    showConfirm('Delete Document', `Delete "${doc.name}"? This cannot be undone.`, async () => {
      const ok = await supabase.deleteDocument(doc.id);
      if (ok) load();
      else showAlert('Error', 'Could not delete document.');
    });
  };

  const handleOpen = (doc) => {
    const { Linking } = require('react-native');
    Linking.openURL(doc.file_url).catch(() => showAlert('Error', 'Could not open this file.'));
  };

  const getFileIcon = (fileType) => {
    if (!fileType) return 'document-outline';
    if (fileType.includes('pdf')) return 'document-text-outline';
    if (fileType.includes('image')) return 'image-outline';
    if (fileType.includes('word') || fileType.includes('doc')) return 'document-outline';
    return 'attach-outline';
  };

  const filteredDocs = documents
    .filter(d => filter === 'all' ? true : filter === 'untagged' ? !d.client_id : d.client_id === filter)
    .filter(d => docSearch ? d.name.toLowerCase().includes(docSearch.toLowerCase()) : true);

  const clientName = (clientId) => clients.find(c => c.id === clientId)?.name || 'Unknown';

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      <View style={{ paddingHorizontal: H, paddingTop: 20, paddingBottom: 12 }}>
        <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 14 }}>
          <View>
            <Text style={{ fontSize: 26, fontWeight: '800', color: C.textDark }}>Documents</Text>
            {preselectedClientName && (
              <Text style={{ fontSize: 13, color: C.primary, fontWeight: '600', marginTop: 2 }}>Uploading to: {preselectedClientName}</Text>
            )}
          </View>
          <TouchableOpacity
            style={[st.addBtn, { opacity: uploading ? 0.6 : 1 }]}
            onPress={handleUpload}
            disabled={uploading}
            activeOpacity={0.8}
          >
            {uploading
              ? <ActivityIndicator color={C.white} size="small" />
              : <>
                  <Ionicons name="cloud-upload-outline" size={16} color={C.white} />
                  <Text style={st.addBtnText}>Upload</Text>
                </>
            }
          </TouchableOpacity>
        </View>

        {/* SEARCH + FILTER */}
        <View style={[st.searchBar, { marginBottom: 8 }]}>
          <Ionicons name="search-outline" size={18} color={C.textMute} style={{ marginRight: 8 }} />
          <TextInput
            style={{ flex: 1, fontSize: 14, color: C.textDark }}
            value={docSearch}
            onChangeText={setDocSearch}
            placeholder="Search documents..."
            placeholderTextColor={C.textMute}
          />
          {docSearch.length > 0 && (
            <TouchableOpacity onPress={() => setDocSearch('')}>
              <Ionicons name="close-circle" size={18} color={C.textMute} />
            </TouchableOpacity>
          )}
        </View>

// ─── EDIT PROFILE SCREEN ──────────────────────────────────────────────────
function EditProfileScreen({ user, profileName, businessName, logoUrl, onBack, onSaved }) {
  const [fullName, setFullName] = useState(profileName || '');
  const [bizName, setBizName] = useState(businessName || '');
  const [logoUri, setLogoUri] = useState(null);
  const [saving, setSaving] = useState(false);

  const handlePickLogo = () => {
    if (Platform.OS === 'web') {
      showAlert('Not supported', 'Logo upload is not available on web. Use the phone app.');
      return;
    }
    Alert.alert(
      'Upload Logo',
      'Choose where to find your logo',
      [
        {
          text: 'Photo Library',
          onPress: async () => {
            try {
              const perm = await ImagePicker.requestMediaLibraryPermissionsAsync();
              if (!perm.granted) { showAlert('Permission needed', 'Please allow photo access.'); return; }
              const mediaType = ImagePicker.MediaTypeOptions?.Images ?? ImagePicker.MediaType?.images ?? 'Images';
              const result = await ImagePicker.launchImageLibraryAsync({
                mediaTypes: mediaType,
                allowsEditing: true,
                aspect: [1, 1],
                quality: 0.7,
                base64: true,
              });
              if (!result.canceled && result.assets?.[0]) setLogoUri(result.assets[0]);
            } catch (e) { showAlert('Error', 'Could not open photo library.'); }
          },
        },
        {
          text: 'Files',
          onPress: async () => {
            try {
              const result = await DocumentPicker.getDocumentAsync({
                type: ['image/png', 'image/jpeg', 'image/jpg'],
                copyToCacheDirectory: true,
                base64: true,
              });
              if (!result.canceled && result.assets?.[0]) {
                setLogoUri({ uri: result.assets[0].uri, base64: result.assets[0].base64, mimeType: result.assets[0].mimeType || 'image/png' });
              }
            } catch (e) { showAlert('Error', 'Could not open files.'); }
          },
        },
        { text: 'Cancel', style: 'cancel' },
      ]
    );
  };

  const handleSave = async () => {
    // Either a personal name OR a business name is enough — not both required.
    if (!fullName.trim() && !bizName.trim()) {
      showAlert('Required', 'Please enter your name or your business name.');
      return;
    }
    setSaving(true);
    const updates = {
      full_name: fullName.trim() || null,
      business_name: bizName.trim() || null,
    };
    if (logoUri?.uri) {
      const logoUrl = await supabase.uploadLogo(user.id, logoUri.uri, logoUri.mimeType || 'image/jpeg');
      if (logoUrl) updates.logo_url = logoUrl;
    }
    const ok = await supabase.updateProfile(user.id, updates);
    setSaving(false);
    if (ok) {
      showAlert('Saved', 'Your profile has been updated.');
      onSaved(fullName.trim().split(' ')[0], bizName.trim(), updates.logo_url || null);
    } else {
      showAlert('Error', 'Could not save profile. Please try again.');
    }
  };

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      <ScreenHeader
        title="Edit Profile"
        onBack={onBack}
        rightAction={
          <TouchableOpacity onPress={handleSave} disabled={saving} style={{ paddingRight: 4 }}>
            {saving ? <ActivityIndicator color={C.primary} size="small" /> : <Text style={{ fontSize: 15, fontWeight: '700', color: C.primary }}>Save</Text>}
          </TouchableOpacity>
        }
      />
      <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
        <ScrollView contentContainerStyle={{ padding: H, paddingBottom: 60 }} keyboardShouldPersistTaps="handled" showsVerticalScrollIndicator={false}>

          {/* LOGO */}
          <View style={{ alignItems: 'center', marginBottom: 28 }}>
            <TouchableOpacity onPress={handlePickLogo} activeOpacity={0.8}>
              {logoUri ? (
                <Image source={{ uri: logoUri.uri }} style={{ width: 90, height: 90, borderRadius: 20, borderWidth: 2, borderColor: C.primaryLight }} />
              ) : logoUrl ? (
                <Image source={{ uri: logoUrl }} style={{ width: 90, height: 90, borderRadius: 20, borderWidth: 2, borderColor: C.primaryLight }} />
              ) : (
                <View style={{ width: 90, height: 90, borderRadius: 20, backgroundColor: C.cardAlt, borderWidth: 2, borderColor: C.primaryLight, alignItems: 'center', justifyContent: 'center' }}>
                  <Avatar name={fullName} size={86} />
                </View>
              )}
              <View style={{ position: 'absolute', bottom: -6, right: -6, width: 28, height: 28, borderRadius: 14, backgroundColor: C.primary, alignItems: 'center', justifyContent: 'center', borderWidth: 2, borderColor: C.white }}>
                <Ionicons name="camera" size={14} color={C.white} />
              </View>
            </TouchableOpacity>
            <Text style={{ fontSize: 12, color: C.textMute, marginTop: 14 }}>Tap to change logo</Text>
          </View>

          {/* PROFILE FIELDS */}
          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={st.sectionLabel}>YOUR INFO</Text>
            <Text style={st.fieldLabel}>Full Name *</Text>
            <TextInput
              style={[st.fieldInput, { marginBottom: 14 }]}
              value={fullName}
              onChangeText={setFullName}
              placeholder="Your full name"
              placeholderTextColor={C.textMute}
              autoCapitalize="words"
            />
            <Text style={st.fieldLabel}>Business Name</Text>
            <TextInput
              style={st.fieldInput}
              value={bizName}
              onChangeText={setBizName}
              placeholder="Your business or brand name"
              placeholderTextColor={C.textMute}
              autoCapitalize="words"
            />
          </View>

          {/* EMAIL — read only */}
          <View style={[st.card, { padding: 16, marginBottom: 24 }]}>
            <Text style={st.sectionLabel}>ACCOUNT</Text>
            <Text style={st.fieldLabel}>Email</Text>
            <View style={[st.fieldInput, { backgroundColor: C.cardAlt }]}>
              <Text style={{ fontSize: 15, color: C.textMute }}>{user?.email}</Text>
            </View>
            <Text style={{ fontSize: 11, color: C.textMute, marginTop: 6 }}>Email cannot be changed here.</Text>
          </View>

          <TouchableOpacity
            style={[st.primaryBtn, { opacity: saving ? 0.7 : 1 }]}
            onPress={handleSave}
            disabled={saving}
            activeOpacity={0.8}
          >
            {saving ? <ActivityIndicator color={C.white} size="small" /> : <Text style={st.primaryBtnText}>Save Profile</Text>}
          </TouchableOpacity>

        </ScrollView>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

// ─── COMMAND BAR SCREEN ────────────────────────────────────────────────────
function CommandBarScreen({ user, businessName, onBack, onNavigate }) {
  const [speakEnabled, setSpeakEnabled] = useState(true);

  const speakResponse = (text) => {
    if (!speakEnabled) return;
    try {
      Speech.stop();
      Speech.speak(text, {
        language: 'en-US',
        pitch: 1.0,
        rate: 0.92,
      });
    } catch (e) {}
  };
  const [input, setInput] = useState('');
  const [messages, setMessages] = useState([]);
  const [loading, setLoading] = useState(false);
  const [pendingAction, setPendingAction] = useState(null);
  const scrollRef = React.useRef(null);

  const quickPrompts = [
    "What needs my attention today?",
    "How's business this month?",
    "Who hasn't paid yet?",
    "What's due this week?",
  ];

  const addMessage = (role, text, action = null, actionData = {}, requiresConfirmation = false) => {
    setMessages(prev => [...prev, { role, text, action, actionData, requiresConfirmation }]);
    setTimeout(() => scrollRef.current?.scrollToEnd({ animated: true }), 100);
    if (role === 'bizhq' && text) speakResponse(text);
  };

  const handleSend = async (commandText = null) => {
    const cmd = (commandText || input).trim();
    if (!cmd || loading) return;
    const limitCheck = await checkFreeLimit(user.id, 'free', 'ai');
    if (!limitCheck.allowed) { showAlert('Upgrade Required', limitCheck.message); return; }
    setInput('');
    addMessage('user', cmd);
    setLoading(true);

    // Build conversation history so the assistant remembers the thread.
    const history = messages
      .filter(m => m.role === 'user' || m.role === 'bizhq')
      .map(m => ({ role: m.role === 'user' ? 'user' : 'assistant', content: m.text }));

    const result = await supabase.callAI(cmd, user.id, history);
    setLoading(false);

    if (result.requiresConfirmation && result.action) {
      setPendingAction(result);
      addMessage('bizhq', result.message, result.action, result.actionData, true);
    } else {
      addMessage('bizhq', result.message, result.action, result.actionData, false);
      if (result.action === 'view_client' && result.actionData?.clientId) {
        setTimeout(() => onNavigate('clientDetail', result.actionData.clientId), 800);
      }
    }
  };

  const handleConfirm = async () => {
    if (!pendingAction) return;
    const { action, actionData } = pendingAction;
    setPendingAction(null);

    if (action === 'mark_paid') {
      setLoading(true);
      const ok = await supabase.updateInvoice(actionData.invoiceId, { status: 'paid', paid_at: new Date().toISOString() });
      setLoading(false);
      addMessage('bizhq', ok
        ? `Done. ${actionData.clientName}'s invoice for ${new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(actionData.amount)} is marked paid.`
        : "Couldn't update that invoice. Try again or mark it paid from the invoice screen."
      );
    } else if (action === 'send_reminder') {
      setLoading(true);
      const client = await supabase.getClient(actionData.clientId);
      if (client?.email) {
        const sent = await supabase.sendReminderEmail(
          actionData.clientName,
          client.email,
          actionData.message,
          actionData.invoiceId,
          actionData.invoiceAmount,
          businessName
        );
        await supabase.logReminder(user.id, actionData.clientId, 'invoice', actionData.invoiceId || 'general', 'email', actionData.message);
        setLoading(false);
        addMessage('bizhq', sent
          ? `Done. Reminder sent to ${actionData.clientName} at ${client.email}.`
          : `Reminder logged but the email could not be sent. Check that ${actionData.clientName} has a valid email address.`
        );
      } else {
        setLoading(false);
        if (user?.id && actionData.clientId) {
          await supabase.logReminder(user.id, actionData.clientId, 'invoice', actionData.invoiceId || 'general', 'email', actionData.message);
        }
        addMessage('bizhq', `${actionData.clientName} does not have an email address on file. Add one to their profile and try again.`);
      }
    } else if (action === 'create_invoice') {
      addMessage('bizhq', `Opening a ready-to-review invoice for ${actionData.clientName}.`);
      // Pre-fill the invoice with what the AI gathered (client + amount + title).
      const prefill = {
        clientId: actionData.clientId,
        title: actionData.suggestedTitle || 'Services',
        amount: actionData.suggestedAmount || null,
      };
      setTimeout(() => onNavigate('newInvoice', prefill), 600);
    }
  };

  const handleDecline = () => {
    setPendingAction(null);
    addMessage('bizhq', "No problem. What else can I help with?");
  };

  return (
    <SafeAreaView style={[st.safe, { backgroundColor: '#A0623A' }]}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />

      {/* HEADER */}
      <View style={{ flexDirection: 'row', alignItems: 'center', paddingHorizontal: H, paddingVertical: 14, borderBottomWidth: 1, borderBottomColor: '#3D2015' }}>
        <TouchableOpacity onPress={onBack} style={{ flexDirection: 'row', alignItems: 'center', marginRight: 12 }}>
          <Ionicons name="chevron-back" size={20} color={C.primaryLight} />
        </TouchableOpacity>
        <View style={{ width: 36, height: 36, borderRadius: 10, backgroundColor: C.primary, alignItems: 'center', justifyContent: 'center', marginRight: 10 }}>
          <Ionicons name="chatbubble-ellipses" size={18} color={C.white} />
        </View>
        <View style={{ flex: 1 }}>
          <Text style={{ fontSize: 16, fontWeight: '700', color: C.white }}>BizHQ AI</Text>
          <Text style={{ fontSize: 11, color: C.primaryLight }}>Your business headquarters</Text>
        </View>
        <TouchableOpacity
          onPress={() => {
            if (speakEnabled) { Speech.stop(); setSpeakEnabled(false); }
            else setSpeakEnabled(true);
          }}
          style={{ width: 36, height: 36, borderRadius: 10, backgroundColor: speakEnabled ? C.primary : '#3D2015', alignItems: 'center', justifyContent: 'center', marginRight: 6 }}
          activeOpacity={0.8}
        >
          <Ionicons name={speakEnabled ? 'volume-high' : 'volume-mute-outline'} size={18} color={speakEnabled ? C.white : '#8B6050'} />
        </TouchableOpacity>
        <TouchableOpacity
          onPress={() => {
            Speech.stop();
            setMessages([]);
            setInput('');
            setPendingAction(null);
          }}
          style={{ width: 36, height: 36, borderRadius: 10, backgroundColor: '#3D2015', alignItems: 'center', justifyContent: 'center' }}
          activeOpacity={0.8}
        >
          <Ionicons name="trash-outline" size={16} color={C.primary} />
        </TouchableOpacity>
      </View>

      {/* MESSAGES */}
      <ScrollView
        ref={scrollRef}
        style={{ flex: 1, backgroundColor: C.background }}
        contentContainerStyle={{ padding: H, paddingBottom: 20 }}
        showsVerticalScrollIndicator={false}
        onContentSizeChange={() => scrollRef.current?.scrollToEnd({ animated: true })}
      >
        {messages.map((msg, idx) => (
          <View key={idx} style={{ marginBottom: 14, alignItems: msg.role === 'user' ? 'flex-end' : 'flex-start' }}>
            {msg.role === 'bizhq' && (
              <View style={{ flexDirection: 'row', alignItems: 'flex-start' }}>
                <View style={{ width: 28, height: 28, borderRadius: 8, backgroundColor: C.primary, alignItems: 'center', justifyContent: 'center', marginRight: 8, marginTop: 2 }}>
                  <Ionicons name="chatbubble-ellipses" size={14} color={C.white} />
                </View>
                <View style={{ flex: 1 }}>
                  <View style={{ backgroundColor: '#2C1810', borderRadius: 12, borderTopLeftRadius: 2, padding: 14, maxWidth: '90%' }}>
                    <Text style={{ fontSize: 14, color: '#FAF0E6', lineHeight: 22 }}>{msg.text}</Text>
                  </View>
                  {msg.requiresConfirmation && pendingAction && (
                    <View style={{ flexDirection: 'row', gap: 8, marginTop: 10 }}>
                      <TouchableOpacity
                        style={{ flex: 1, backgroundColor: C.primary, borderRadius: 10, paddingVertical: 10, alignItems: 'center' }}
                        onPress={handleConfirm}
                        activeOpacity={0.8}
                      >
                        <Text style={{ fontSize: 13, fontWeight: '700', color: C.white }}>Yes, do it</Text>
                      </TouchableOpacity>
                      <TouchableOpacity
                        style={{ flex: 1, backgroundColor: '#3D2015', borderRadius: 10, paddingVertical: 10, alignItems: 'center' }}
                        onPress={handleDecline}
                        activeOpacity={0.8}
                      >
                        <Text style={{ fontSize: 13, fontWeight: '600', color: '#FAF0E6' }}>Never mind</Text>
                      </TouchableOpacity>
                    </View>
                  )}
                  {msg.action === 'view_client' && msg.actionData?.clientId && !msg.requiresConfirmation && (
                    <TouchableOpacity
                      style={{ marginTop: 8, flexDirection: 'row', alignItems: 'center' }}
                      onPress={() => onNavigate('clientDetail', msg.actionData.clientId)}
                    >
                      <Text style={{ fontSize: 13, color: C.primaryLight, fontWeight: '600' }}>View {msg.actionData.clientName}'s profile</Text>
                      <Ionicons name="chevron-forward" size={14} color={C.primaryLight} style={{ marginLeft: 4 }} />
                    </TouchableOpacity>
                  )}
                </View>
              </View>
            )}
            {msg.role === 'user' && (
              <View style={{ backgroundColor: C.primary, borderRadius: 12, borderTopRightRadius: 2, padding: 14, maxWidth: '80%' }}>
                <Text style={{ fontSize: 14, color: C.white, lineHeight: 22 }}>{msg.text}</Text>
              </View>
            )}
          </View>
        ))}
        {loading && (
          <View style={{ flexDirection: 'row', alignItems: 'center', marginBottom: 14 }}>
            <View style={{ width: 28, height: 28, borderRadius: 8, backgroundColor: C.primary, alignItems: 'center', justifyContent: 'center', marginRight: 8 }}>
              <Ionicons name="chatbubble-ellipses" size={14} color={C.white} />
            </View>
            <View style={{ backgroundColor: '#2C1810', borderRadius: 12, borderTopLeftRadius: 2, padding: 14 }}>
              <ActivityIndicator color={C.primaryLight} size="small" />
            </View>
          </View>
        )}
      </ScrollView>

      {/* QUICK PROMPTS */}
      {messages.length === 0 && (
        <View style={{ paddingHorizontal: H, paddingBottom: 8, flexDirection: 'row', flexWrap: 'wrap', gap: 6 }}>
          {quickPrompts.map((prompt) => (
            <TouchableOpacity
              key={prompt}
              style={{ backgroundColor: '#2C1810', borderRadius: 14, paddingHorizontal: 10, paddingVertical: 5, borderWidth: 1, borderColor: '#8B4A28' }}
              onPress={() => handleSend(prompt)}
              activeOpacity={0.7}
            >
              <Text style={{ fontSize: 11, color: '#FFFFFF', fontWeight: '600' }} numberOfLines={1}>{prompt}</Text>
            </TouchableOpacity>
          ))}
        </View>
      )}

      {/* INPUT */}
      <KeyboardAvoidingView behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
        <View style={{ flexDirection: 'row', alignItems: 'center', paddingHorizontal: H, paddingVertical: 12, borderTopWidth: 1, borderTopColor: '#3D2015', gap: 8 }}>
          <TextInput
            style={{
              flex: 1, backgroundColor: '#2C1810', borderRadius: 24, paddingHorizontal: 16, paddingVertical: 12,
              fontSize: 14, color: '#FAF0E6', borderWidth: 1, borderColor: '#8B4A28', maxHeight: 100,
            }}
            value={input}
            onChangeText={setInput}
            placeholder="Type or use mic on keyboard..."
            placeholderTextColor="#B09080"
            multiline
            returnKeyType="send"
            onSubmitEditing={() => handleSend()}
          />
          <TouchableOpacity
            style={{
              width: 44, height: 44, borderRadius: 22,
              backgroundColor: input.trim() ? C.primary : '#8B4A28',
              alignItems: 'center', justifyContent: 'center',
            }}
            onPress={() => handleSend()}
            disabled={!input.trim() || loading}
            activeOpacity={0.8}
          >
            <Ionicons name="send" size={18} color={C.white} />
          </TouchableOpacity>
        </View>

      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

  const [startDate, setStartDate] = useState('');
  const [endDate, setEndDate] = useState('');
  const [notes, setNotes] = useState('');
  const [saving, setSaving] = useState(false);
  const [sending, setSending] = useState(false);

  const defaultBody = `SERVICE AGREEMENT

This agreement is entered into between the service provider and the client identified below.

SCOPE OF WORK
[Describe the services to be provided]

PAYMENT TERMS
Total Amount: $[Amount]
Deposit: $[Deposit] due upon signing
Balance: Due upon completion

TIMELINE
Start Date: [Start Date]
End Date / Delivery: [End Date]

REVISIONS
[Number] rounds of revisions included. Additional revisions billed at $[Rate]/hour.

CANCELLATION
Either party may cancel with [X] days written notice. Deposit is non-refundable.

AGREEMENT
By signing below, both parties agree to the terms outlined in this contract.

Client Signature: _______________________  Date: ___________`;

  const [body, setBody] = useState(defaultBody);

  useEffect(() => {
    // clientId may be a plain id (string) OR a prefill object { clientId, title, amount }.
    const prefill = (clientId && typeof clientId === 'object') ? clientId : null;
    const effectiveClientId = prefill ? prefill.clientId : clientId;
    if (prefill) {
      if (prefill.title) setTitle(prefill.title);
      if (prefill.amount) setAmount(String(prefill.amount));
      if (prefill.deposit) setDepositAmount(String(prefill.deposit));
    }
    supabase.getClients(user.id).then(data => {
      setClients(data);
      if (effectiveClientId) {
        const found = data.find(c => c.id === effectiveClientId);
        if (found) setSelectedClient(found);
      }
    });
  }, []);

  const handleSave = async (sendNow = false) => {
    if (!selectedClient) { showAlert('Required', 'Please select a client.'); return; }
    if (!title.trim()) { showAlert('Required', 'Please enter a contract title.'); return; }
    if (!amount || isNaN(parseFloat(amount))) { showAlert('Required', 'Please enter the contract amount.'); return; }
    if (sendNow && !selectedClient.email) {
      showAlert('No Email', 'This client does not have an email address. Add one to their profile before sending for signature.');
      return;
    }
    const limitCheck = await checkFreeLimit(user.id, 'free', 'contracts');
    if (!limitCheck.allowed) { showAlert('Upgrade Required', limitCheck.message); return; }

    sendNow ? setSending(true) : setSaving(true);

    const contract = {
      user_id: user.id,
      client_id: selectedClient.id,
      title: title.trim(),
      body: body.trim(),
      amount: parseFloat(amount),
      deposit_amount: depositAmount ? parseFloat(depositAmount) : null,
      start_date: startDate || null,
      end_date: endDate || null,
      notes: notes.trim() || null,
      status: 'draft',
    };

    const result = await supabase.addContract(contract);
    if (!result) {
      setSaving(false); setSending(false);
      showAlert('Error', 'Could not save contract. Please try again.');
      return;
    }

    if (sendNow) {
      const boldResult = await supabase.sendContractToBoldSign(
        result.id, title.trim(), body.trim(),
        selectedClient.name, selectedClient.email
      );
      if (boldResult?.documentId) {
        await supabase.updateContract(result.id, {
          status: 'sent',
          sent_at: new Date().toISOString(),
          boldsign_document_id: boldResult.documentId,
          boldsign_signing_url: boldResult.signingUrl || null,
        });
        showAlert('Sent', `Contract sent to ${selectedClient.name} for signature.`);
      } else {
        showAlert('Saved', 'Contract saved. BoldSign could not send it — you can retry from the contract detail screen.');
      }
    }

    setSaving(false); setSending(false);
    onSaved(result.id);
  };

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      <ScreenHeader
        title="New Contract"
        onBack={onBack}
        rightAction={
          <TouchableOpacity onPress={() => handleSave(false)} disabled={saving || sending} style={{ paddingRight: 4 }}>
            {saving ? <ActivityIndicator color={C.primary} size="small" /> : <Text style={{ fontSize: 15, fontWeight: '700', color: C.primary }}>Save</Text>}
          </TouchableOpacity>
        }
      />
      <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
        <ScrollView contentContainerStyle={{ padding: H, paddingBottom: 60 }} keyboardShouldPersistTaps="handled" showsVerticalScrollIndicator={false}>

          {/* CLIENT */}
          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={st.sectionLabel}>CLIENT</Text>
            <TouchableOpacity
              style={[st.fieldInput, { flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between' }]}
              onPress={() => setShowClientPicker(!showClientPicker)}
              activeOpacity={0.7}
            >
              <Text style={{ fontSize: 15, color: selectedClient ? C.textDark : C.textMute }}>
                {selectedClient ? selectedClient.name : 'Select a client...'}
              </Text>
              <Ionicons name={showClientPicker ? 'chevron-up' : 'chevron-down'} size={18} color={C.textMute} />
            </TouchableOpacity>
            {showClientPicker && (
              <View style={{ marginTop: 8, borderWidth: 1, borderColor: C.border, borderRadius: 10, overflow: 'hidden' }}>
                {clients.map((c, idx) => (
                  <TouchableOpacity
                    key={c.id}
                    style={{ padding: 14, borderTopWidth: idx === 0 ? 0 : 1, borderTopColor: C.borderLight, backgroundColor: selectedClient?.id === c.id ? C.cardAlt : C.white }}
                    onPress={() => { setSelectedClient(c); setShowClientPicker(false); }}
                    activeOpacity={0.7}
                  >
                    <Text style={{ fontSize: 14, fontWeight: selectedClient?.id === c.id ? '700' : '400', color: C.textDark }}>{c.name}</Text>
                    {c.email ? <Text style={{ fontSize: 12, color: C.textMute }}>{c.email}</Text> : <Text style={{ fontSize: 12, color: C.overdue }}>No email — required for signature</Text>}
                  </TouchableOpacity>
                ))}
              </View>
            )}
          </View>

          {/* CONTRACT DETAILS */}
          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={st.sectionLabel}>CONTRACT DETAILS</Text>
            <Text style={st.fieldLabel}>Title *</Text>
            <TextInput style={[st.fieldInput, { marginBottom: 14 }]} value={title} onChangeText={setTitle} placeholder="e.g. 3-Month Coaching Agreement" placeholderTextColor={C.textMute} autoCapitalize="words" />
            <View style={{ flexDirection: 'row', gap: 10 }}>
              <View style={{ flex: 1 }}>
                <Text style={st.fieldLabel}>Total Amount *</Text>
                <View style={{ flexDirection: 'row', alignItems: 'center' }}>
                  <Text style={{ fontSize: 16, fontWeight: '600', color: C.textMid, marginRight: 6 }}>$</Text>
                  <TextInput style={[st.fieldInput, { flex: 1 }]} value={amount} onChangeText={setAmount} placeholder="0.00" placeholderTextColor={C.textMute} keyboardType="decimal-pad" />
                </View>
              </View>
              <View style={{ flex: 1 }}>
                <Text style={st.fieldLabel}>Deposit</Text>
                <View style={{ flexDirection: 'row', alignItems: 'center' }}>
                  <Text style={{ fontSize: 16, fontWeight: '600', color: C.textMid, marginRight: 6 }}>$</Text>
                  <TextInput style={[st.fieldInput, { flex: 1 }]} value={depositAmount} onChangeText={setDepositAmount} placeholder="Optional" placeholderTextColor={C.textMute} keyboardType="decimal-pad" />
                </View>
              </View>
            </View>
            <View style={{ flexDirection: 'row', gap: 10, marginTop: 14 }}>
              <View style={{ flex: 1 }}>
                <Text style={st.fieldLabel}>Start Date</Text>
                <TextInput style={st.fieldInput} value={startDate} onChangeText={setStartDate} placeholder="YYYY-MM-DD" placeholderTextColor={C.textMute} keyboardType="numbers-and-punctuation" maxLength={10} />
              </View>
              <View style={{ flex: 1 }}>
                <Text style={st.fieldLabel}>End Date</Text>
                <TextInput style={st.fieldInput} value={endDate} onChangeText={setEndDate} placeholder="YYYY-MM-DD" placeholderTextColor={C.textMute} keyboardType="numbers-and-punctuation" maxLength={10} />
              </View>
            </View>
          </View>

          {/* CONTRACT BODY */}
          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={st.sectionLabel}>CONTRACT BODY</Text>
            <Text style={{ fontSize: 12, color: C.textMute, marginBottom: 10 }}>Edit the contract text below. This is what your client will sign.</Text>
            <TextInput
              style={[st.fieldInput, { height: 280, textAlignVertical: 'top', paddingTop: 12, fontSize: 13, lineHeight: 20 }]}
              value={body}
              onChangeText={setBody}
              multiline
              autoCorrect={false}
            />
          </View>

          {/* NOTES */}
          <View style={[st.card, { padding: 16, marginBottom: 24 }]}>
            <Text style={st.sectionLabel}>INTERNAL NOTES</Text>
            <TextInput style={[st.fieldInput, { height: 80, textAlignVertical: 'top', paddingTop: 12 }]} value={notes} onChangeText={setNotes} placeholder="Private notes — not shown to client" placeholderTextColor={C.textMute} multiline />
          </View>

          {/* ACTIONS */}
          <TouchableOpacity style={[st.primaryBtn, { marginBottom: 12, opacity: (saving || sending) ? 0.7 : 1 }]} onPress={() => handleSave(true)} disabled={saving || sending} activeOpacity={0.8}>
            {sending ? <ActivityIndicator color={C.white} size="small" /> : <Text style={st.primaryBtnText}>Save & Send for Signature</Text>}
          </TouchableOpacity>
          <TouchableOpacity style={[st.outlineBtn, { opacity: (saving || sending) ? 0.7 : 1 }]} onPress={() => handleSave(false)} disabled={saving || sending} activeOpacity={0.8}>
            {saving ? <ActivityIndicator color={C.primary} size="small" /> : <Text style={st.outlineBtnText}>Save as Draft</Text>}
          </TouchableOpacity>

        </ScrollView>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

// ─── CONTRACT DETAIL SCREEN ────────────────────────────────────────────────
function ContractDetailScreen({ user, contractId, onBack, onNavigate }) {
  const [contract, setContract] = useState(null);
  const [loading, setLoading] = useState(true);
  const [updating, setUpdating] = useState(false);
  const [checkingStatus, setCheckingStatus] = useState(false);

  const [clientEmail, setClientEmail] = useState(null);
  const [clientName, setClientName] = useState(null);

  const load = async () => {
    setLoading(true);
    const c = await supabase.getContract(contractId);
    setContract(c);
    // Fetch client email — it lives on bp_clients, not bp_contracts
    if (c?.client_id) {
      const client = await supabase.getClient(c.client_id);
      if (client) {
        setClientEmail(client.email || null);
        setClientName(client.name || null);
      }
    }
    setLoading(false);
  };

  useEffect(() => { if (contractId) load(); }, [contractId]);

  const handleSendForSignature = async () => {
    if (!clientEmail) { showAlert('No Email', 'This client does not have an email address. Edit their profile to add one.'); return; }
    setUpdating(true);
    const result = await supabase.sendContractToBoldSign(
      contract.id, contract.title, contract.body,
      clientName || 'Client', clientEmail
    );
    if (result?.documentId) {
      await supabase.updateContract(contract.id, {
        status: 'sent',
        sent_at: new Date().toISOString(),
        boldsign_document_id: result.documentId,
        boldsign_signing_url: result.signingUrl || null,
      });
      showAlert('Sent', 'Contract sent for signature.');
      load();
    } else {
      showAlert('Error', 'Could not send contract via BoldSign. Please try again.');
    }
    setUpdating(false);
  };

  const handleCheckStatus = async () => {
    if (!contract?.boldsign_document_id) return;
    setCheckingStatus(true);
    const result = await supabase.checkContractStatus(contract.boldsign_document_id);
    setCheckingStatus(false);
    if (result?.isSigned) {
      await supabase.updateContract(contract.id, { status: 'signed', signed_at: new Date().toISOString() });
      showAlert('Signed!', 'The contract has been signed by your client.');
      load();
    } else if (result?.isDeclined) {
      await supabase.updateContract(contract.id, { status: 'declined' });
      showAlert('Declined', 'The client declined to sign this contract.');
      load();
    } else {
      showAlert('Still Pending', `Current status: ${result?.status || 'Unknown'}. The client has not signed yet.`);
    }
  };

  const handleOpenSigningLink = () => {
    if (!contract?.boldsign_signing_url) { showAlert('No Link', 'No signing link available.'); return; }
    const { Linking } = require('react-native');
    Linking.openURL(contract.boldsign_signing_url);
  };

  if (loading) return (
    <SafeAreaView style={st.safe}>
      <ScreenHeader title="Contract" onBack={onBack} />
      <ActivityIndicator color={C.primary} style={{ marginTop: 60 }} />
    </SafeAreaView>
  );

  if (!contract) return (
    <SafeAreaView style={st.safe}>
      <ScreenHeader title="Contract" onBack={onBack} />
      <View style={{ alignItems: 'center', paddingTop: 60 }}><Text style={{ color: C.textMute }}>Contract not found.</Text></View>
    </SafeAreaView>
  );

  const isSigned = contract.status === 'signed';

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      <ScreenHeader title="Contract" onBack={onBack} />
      <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={{ padding: H, paddingBottom: 40 }}>

        {/* HERO */}
        <View style={{ backgroundColor: C.primary, borderRadius: 16, padding: 20, marginBottom: 16, alignItems: 'center' }}>
          <Text style={{ fontSize: 11, color: C.primaryPale, fontWeight: '700', letterSpacing: 1, marginBottom: 6 }}>CONTRACT VALUE</Text>
          <Text style={{ fontSize: 36, fontWeight: '800', color: C.white, marginBottom: 10 }}>{fmt(contract.amount)}</Text>
          <Badge status={contract.status} />
          <Text style={{ fontSize: 14, color: C.primaryPale, marginTop: 8, textAlign: 'center' }}>{contract.title}</Text>
        </View>

        {/* DETAILS */}
        <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
          <Text style={[st.sectionLabel, { marginBottom: 12 }]}>DETAILS</Text>
          {[
            contract.deposit_amount && { label: 'Deposit', value: fmt(contract.deposit_amount) },
            contract.start_date && { label: 'Start Date', value: contract.start_date },
            contract.end_date && { label: 'End Date', value: contract.end_date },
            contract.sent_at && { label: 'Sent', value: new Date(contract.sent_at).toLocaleDateString() },
            contract.signed_at && { label: 'Signed', value: new Date(contract.signed_at).toLocaleDateString() },
            { label: 'Created', value: new Date(contract.created_at).toLocaleDateString() },
          ].filter(Boolean).map(row => (
            <View key={row.label} style={{ flexDirection: 'row', justifyContent: 'space-between', paddingVertical: 8, borderTopWidth: 1, borderTopColor: C.borderLight }}>
              <Text style={{ fontSize: 13, color: C.textMute }}>{row.label}</Text>
              <Text style={{ fontSize: 13, color: C.textDark, fontWeight: '600' }}>{row.value}</Text>
            </View>
          ))}
        </View>

        {/* CONTRACT BODY PREVIEW */}
        <View style={[st.card, { padding: 16, marginBottom: 16 }]}>
          <Text style={[st.sectionLabel, { marginBottom: 10 }]}>CONTRACT</Text>
          <Text style={{ fontSize: 13, color: C.textDark, lineHeight: 20 }} numberOfLines={8}>{contract.body}</Text>
        </View>

        {/* ACTIONS */}
        {updating || checkingStatus
          ? <ActivityIndicator color={C.primary} style={{ marginVertical: 16 }} />
          : isSigned
            ? <View style={[st.card, { padding: 16, alignItems: 'center', borderColor: C.paid, borderWidth: 1.5, marginBottom: 12 }]}>
                <Ionicons name="checkmark-circle" size={28} color={C.paid} style={{ marginBottom: 6 }} />
                <Text style={{ fontSize: 15, fontWeight: '700', color: C.paid }}>Contract Signed</Text>
                <TouchableOpacity style={[st.primaryBtn, { marginTop: 14, width: '100%' }]} onPress={() => onNavigate('newInvoice', { clientId: contract.client_id, title: contract.title, amount: contract.amount })} activeOpacity={0.8}>
                  <Text style={st.primaryBtnText}>Create Invoice</Text>
                </TouchableOpacity>
              </View>
            : contract.status === 'draft'
              ? <TouchableOpacity style={st.primaryBtn} onPress={handleSendForSignature} activeOpacity={0.8}>
                  <Text style={st.primaryBtnText}>Send for Signature</Text>
                </TouchableOpacity>
              : contract.status === 'sent'
                ? <View style={{ gap: 10 }}>
                    <TouchableOpacity style={st.primaryBtn} onPress={handleCheckStatus} activeOpacity={0.8}>
                      <Text style={st.primaryBtnText}>Check Signature Status</Text>
                    </TouchableOpacity>
                    {contract.boldsign_signing_url && (
                      <TouchableOpacity style={[st.outlineBtn, { borderColor: C.primary }]} onPress={handleOpenSigningLink} activeOpacity={0.8}>
                        <Text style={[st.outlineBtnText, { color: C.primary }]}>Open Signing Link</Text>
                      </TouchableOpacity>
                    )}
                  </View>
                : null
        }

        <TouchableOpacity
          style={{ marginTop: 24, alignItems: 'center', paddingVertical: 12 }}
          onPress={() => showConfirm('Archive Contract', 'This moves the contract out of your active list. You can keep it for your records.', async () => {
            const ok = await supabase.archiveContract(contract.id);
            if (ok) { showAlert('Archived', 'Contract moved to archive.'); onBack(); }
            else { showAlert('Error', 'Could not archive. Try again.'); }
          })}
          activeOpacity={0.7}
        >
          <View style={{ flexDirection: 'row', alignItems: 'center', justifyContent: 'center' }}>
            <Ionicons name="archive-outline" size={16} color={C.primary} style={{ marginRight: 6 }} />
            <Text style={{ fontSize: 14, color: C.primary, fontWeight: '700' }}>Archive this contract</Text>
          </View>
        </TouchableOpacity>

      </ScrollView>
    </SafeAreaView>
  );
}

// ─── EDIT INVOICE SCREEN ──────────────────────────────────────────────────
function EditInvoiceScreen({ user, invoiceId, onBack, onSaved }) {
  const [title, setTitle] = useState('');
  const [lineItems, setLineItems] = useState([{ description: '', quantity: '1', unitPrice: '' }]);
  const [depositApplied, setDepositApplied] = useState('');
  const [dueDate, setDueDate] = useState('');
  const [notes, setNotes] = useState('');
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    const load = async () => {
      const [inv, items] = await Promise.all([
        supabase.getInvoice(invoiceId),
        supabase.getLineItems(invoiceId),
      ]);
      if (inv) {
        setTitle(inv.title || '');
        setDepositApplied(inv.deposit_applied ? String(inv.deposit_applied) : '');
        setDueDate(inv.due_date || '');
        setNotes(inv.notes || '');
      }
      if (items && items.length > 0) {
        setLineItems(items.map(i => ({
          description: i.description,
          quantity: String(i.quantity),
          unitPrice: String(i.unit_price),
        })));
      }
      setLoading(false);
    };
    load();
  }, [invoiceId]);

  const addLineItem = () => setLineItems([...lineItems, { description: '', quantity: '1', unitPrice: '' }]);
  const removeLineItem = (idx) => { if (lineItems.length > 1) setLineItems(lineItems.filter((_, i) => i !== idx)); };
  const updateLineItem = (idx, field, val) => {
    const updated = [...lineItems];
    updated[idx][field] = val;
    setLineItems(updated);
  };

  const subtotal = lineItems.reduce((sum, item) => {
    return sum + ((parseFloat(item.quantity) || 0) * (parseFloat(item.unitPrice) || 0));
  }, 0);
  const deposit = parseFloat(depositApplied) || 0;
  const totalDue = Math.max(0, subtotal - deposit);

  const handleSave = async () => {
    if (lineItems.every(i => !i.description.trim() || !i.unitPrice)) {
      showAlert('Required', 'Please add at least one line item.');
      return;
    }
    setSaving(true);
    const updates = {
      title: title.trim() || null,
      subtotal,
      deposit_applied: deposit,
      total_due: totalDue,
      due_date: dueDate || null,
      notes: notes.trim() || null,
    };
    const ok = await supabase.updateInvoice(invoiceId, updates);
    if (!ok) { setSaving(false); showAlert('Error', 'Could not save changes.'); return; }
    onSaved(invoiceId);
    setSaving(false);
  };

  if (loading) return (
    <SafeAreaView style={st.safe}>
      <ScreenHeader title="Edit Invoice" onBack={onBack} />
      <ActivityIndicator color={C.primary} style={{ marginTop: 60 }} />
    </SafeAreaView>
  );

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor={C.background} />
      <ScreenHeader
        title="Edit Invoice"
        onBack={onBack}
        rightAction={
          <TouchableOpacity onPress={handleSave} disabled={saving} style={{ paddingRight: 4 }}>
            {saving ? <ActivityIndicator color={C.primary} size="small" /> : <Text style={{ fontSize: 15, fontWeight: '700', color: C.primary }}>Save</Text>}
          </TouchableOpacity>
        }
      />
      <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
        <ScrollView contentContainerStyle={{ padding: H, paddingBottom: 60 }} keyboardShouldPersistTaps="handled" showsVerticalScrollIndicator={false}>

// ─── NEW INVOICE SCREEN ────────────────────────────────────────────────────
function NewInvoiceScreen({ user, clientId, onBack, onSaved, onNavigate }) {
  const [clients, setClients] = useState([]);
  const [selectedClient, setSelectedClient] = useState(null);
  const [showClientPicker, setShowClientPicker] = useState(false);
  const [title, setTitle] = useState('');
  const [lineItems, setLineItems] = useState([{ description: '', quantity: '1', unitPrice: '' }]);
  const [depositApplied, setDepositApplied] = useState('');
  const [dueDate, setDueDate] = useState('');
  const [notes, setNotes] = useState('');
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    // clientId may be a plain id (string) OR a prefill object { clientId, title, amount }.
    const prefill = (clientId && typeof clientId === 'object') ? clientId : null;
    const effectiveClientId = prefill ? prefill.clientId : clientId;
    if (prefill) {
      if (prefill.title) setTitle(prefill.title);
      if (prefill.timeEntries && prefill.timeEntries.length > 0) {
        // Billing tracked time: each entry becomes a line item.
        setLineItems(prefill.timeEntries.map(t => ({
          description: t.description || 'Work',
          quantity: String(t.hours),
          unitPrice: t.hourly_rate ? String(t.hourly_rate) : '0',
        })));
        setTitle(prefill.title || 'Time & Services');
      } else if (prefill.amount) {
        setLineItems([{ description: prefill.title || 'Services', quantity: '1', unitPrice: String(prefill.amount) }]);
      }
    }
    supabase.getClients(user.id).then(data => {
      setClients(data);
      if (effectiveClientId) {
        const found = data.find(c => c.id === effectiveClientId);
        if (found) setSelectedClient(found);
      }
    });
  }, []);

  const addLineItem = () => setLineItems([...lineItems, { description: '', quantity: '1', unitPrice: '' }]);
  const removeLineItem = (idx) => { if (lineItems.length > 1) setLineItems(lineItems.filter((_, i) => i !== idx)); };
  const updateLineItem = (idx, field, val) => {
    const updated = [...lineItems];
    updated[idx][field] = val;
    setLineItems(updated);
  };

  const subtotal = lineItems.reduce((sum, item) => {
    const qty = parseFloat(item.quantity) || 0;
    const price = parseFloat(item.unitPrice) || 0;
    return sum + (qty * price);
  }, 0);
  const deposit = parseFloat(depositApplied) || 0;
  const totalDue = Math.max(0, subtotal - deposit);

  const handleSave = async () => {
    if (!selectedClient) { showAlert('Required', 'Please select a client.'); return; }
    if (lineItems.every(i => !i.description.trim() || !i.unitPrice)) { showAlert('Required', 'Please add at least one line item with a description and price.'); return; }
    const limitCheck = await checkFreeLimit(user.id, user.subscriptionStatus || 'free', 'invoices');
    if (!limitCheck.allowed) { showAlert('Upgrade Required', limitCheck.message); return; }
    setSaving(true);

    const invoiceId = `BPI-${Date.now()}`;
    const invoice = {
      id: invoiceId,
      user_id: user.id,
      client_id: selectedClient.id,
      title: title.trim() || null,
      subtotal,
      deposit_applied: deposit,
      total_due: totalDue,
      status: 'pending',
      due_date: dueDate || null,
      notes: notes.trim() || null,
      archived: false,
    };

    const result = await supabase.addInvoice(invoice);
    if (!result) { setSaving(false); showAlert('Error', 'Could not save invoice. Please try again.'); return; }

    // Save line items
    const items = lineItems
      .filter(i => i.description.trim() && i.unitPrice)
      .map(i => ({
        invoice_id: invoiceId,
        description: i.description.trim(),
        quantity: parseFloat(i.quantity) || 1,
        unit_price: parseFloat(i.unitPrice),
        amount: (parseFloat(i.quantity) || 1) * parseFloat(i.unitPrice),
      }));
    if (items.length > 0) await supabase.addLineItems(items);

    // Generate Stripe payment link
    const paymentUrl = await supabase.createPaymentLink(invoiceId, selectedClient.name, totalDue);
    if (paymentUrl) await supabase.updateInvoice(invoiceId, { payment_link: paymentUrl });

    setSaving(false);
    onSaved(invoiceId);
  };

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      <ScreenHeader
        title="New Invoice"
        onBack={onBack}
        rightAction={
          <TouchableOpacity onPress={handleSave} disabled={saving} style={{ paddingRight: 4 }}>
            {saving ? <ActivityIndicator color={C.primary} size="small" /> : <Text style={{ fontSize: 15, fontWeight: '700', color: C.primary }}>Save</Text>}
          </TouchableOpacity>
        }
      />
      <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
        <ScrollView contentContainerStyle={{ padding: H, paddingBottom: 60 }} keyboardShouldPersistTaps="handled" showsVerticalScrollIndicator={false}>

          {/* CLIENT */}
          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={st.sectionLabel}>CLIENT</Text>
            <TouchableOpacity
              style={[st.fieldInput, { flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between' }]}
              onPress={() => setShowClientPicker(!showClientPicker)}
              activeOpacity={0.7}
            >
              <Text style={{ fontSize: 15, color: selectedClient ? C.textDark : C.textMute }}>
                {selectedClient ? selectedClient.name : 'Select a client...'}
              </Text>
              <Ionicons name={showClientPicker ? 'chevron-up' : 'chevron-down'} size={18} color={C.textMute} />
            </TouchableOpacity>
            {showClientPicker && (
              <View style={{ marginTop: 8, borderWidth: 1, borderColor: C.border, borderRadius: 10, overflow: 'hidden' }}>
                {clients.map((c, idx) => (
                  <TouchableOpacity
                    key={c.id}
                    style={{ padding: 14, borderTopWidth: idx === 0 ? 0 : 1, borderTopColor: C.borderLight, backgroundColor: selectedClient?.id === c.id ? C.cardAlt : C.white }}
                    onPress={() => { setSelectedClient(c); setShowClientPicker(false); }}
                    activeOpacity={0.7}
                  >
                    <Text style={{ fontSize: 14, fontWeight: selectedClient?.id === c.id ? '700' : '400', color: C.textDark }}>{c.name}</Text>
                    {c.company ? <Text style={{ fontSize: 12, color: C.textMute }}>{c.company}</Text> : null}
                  </TouchableOpacity>
                ))}
              </View>
            )}
            <Text style={[st.fieldLabel, { marginTop: 14 }]}>Invoice Title</Text>
            <TextInput style={st.fieldInput} value={title} onChangeText={setTitle} placeholder="e.g. Month 1 — Coaching Sessions" placeholderTextColor={C.textMute} autoCapitalize="words" />
          </View>

          {/* LINE ITEMS */}
          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 12 }}>
              <Text style={st.sectionLabel}>LINE ITEMS</Text>
              <TouchableOpacity onPress={addLineItem} style={{ flexDirection: 'row', alignItems: 'center' }}>
                <Ionicons name="add-circle-outline" size={18} color={C.primary} />
                <Text style={{ fontSize: 13, color: C.primary, fontWeight: '600', marginLeft: 4 }}>Add</Text>
              </TouchableOpacity>
            </View>
            {lineItems.map((item, idx) => (
              <View key={idx} style={{ marginBottom: 16, paddingBottom: 16, borderBottomWidth: idx < lineItems.length - 1 ? 1 : 0, borderBottomColor: C.borderLight }}>
                <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 8 }}>
                  <Text style={[st.fieldLabel, { marginBottom: 0 }]}>Item {idx + 1}</Text>
                  {lineItems.length > 1 && (
                    <TouchableOpacity onPress={() => removeLineItem(idx)}>
                      <Ionicons name="trash-outline" size={16} color={C.overdue} />
                    </TouchableOpacity>
                  )}
                </View>
                <TextInput
                  style={[st.fieldInput, { marginBottom: 8 }]}
                  value={item.description}
                  onChangeText={val => updateLineItem(idx, 'description', val)}
                  placeholder="Description"
                  placeholderTextColor={C.textMute}
                />
                <View style={{ flexDirection: 'row', gap: 8 }}>
                  <View style={{ flex: 1 }}>
                    <Text style={[st.fieldLabel, { fontSize: 11 }]}>Qty</Text>
                    <TextInput
                      style={st.fieldInput}
                      value={item.quantity}
                      onChangeText={val => updateLineItem(idx, 'quantity', val)}
                      keyboardType="decimal-pad"
                      placeholder="1"
                      placeholderTextColor={C.textMute}
                    />
                  </View>
                  <View style={{ flex: 2 }}>
                    <Text style={[st.fieldLabel, { fontSize: 11 }]}>Unit Price ($)</Text>
                    <TextInput
                      style={st.fieldInput}
                      value={item.unitPrice}
                      onChangeText={val => updateLineItem(idx, 'unitPrice', val)}
                      keyboardType="decimal-pad"
                      placeholder="0.00"
                      placeholderTextColor={C.textMute}
                    />
                  </View>
                  <View style={{ flex: 1.5, justifyContent: 'flex-end' }}>
                    <Text style={[st.fieldLabel, { fontSize: 11 }]}>Amount</Text>
                    <View style={[st.fieldInput, { justifyContent: 'center' }]}>
                      <Text style={{ fontSize: 14, fontWeight: '600', color: C.textDark }}>
                        {fmt((parseFloat(item.quantity) || 0) * (parseFloat(item.unitPrice) || 0))}
                      </Text>
                    </View>
                  </View>
                </View>
              </View>
            ))}

            {/* TOTALS */}
            <View style={{ borderTopWidth: 1, borderTopColor: C.border, paddingTop: 12, gap: 6 }}>
              <View style={{ flexDirection: 'row', justifyContent: 'space-between' }}>
                <Text style={{ fontSize: 13, color: C.textMute }}>Subtotal</Text>
                <Text style={{ fontSize: 13, fontWeight: '600', color: C.textDark }}>{fmt(subtotal)}</Text>
              </View>
              <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center' }}>
                <Text style={{ fontSize: 13, color: C.textMute }}>Deposit Applied ($)</Text>
                <TextInput
                  style={[st.fieldInput, { width: 100, textAlign: 'right', paddingVertical: 6 }]}
                  value={depositApplied}
                  onChangeText={setDepositApplied}
                  keyboardType="decimal-pad"
                  placeholder="0.00"
                  placeholderTextColor={C.textMute}
                />
              </View>
              <View style={{ flexDirection: 'row', justifyContent: 'space-between', paddingTop: 8, borderTopWidth: 1, borderTopColor: C.border }}>
                <Text style={{ fontSize: 15, fontWeight: '700', color: C.textDark }}>Total Due</Text>
                <Text style={{ fontSize: 15, fontWeight: '800', color: C.primary }}>{fmt(totalDue)}</Text>
              </View>
            </View>
          </View>

          {/* DETAILS */}
          <View style={[st.card, { padding: 16, marginBottom: 24 }]}>
            <Text style={st.sectionLabel}>DETAILS</Text>
            <Text style={st.fieldLabel}>Due Date</Text>
            <TextInput style={[st.fieldInput, { marginBottom: 14 }]} value={dueDate} onChangeText={setDueDate} placeholder="YYYY-MM-DD" placeholderTextColor={C.textMute} keyboardType="numbers-and-punctuation" maxLength={10} />
            <Text style={st.fieldLabel}>Internal Notes</Text>
            <TextInput style={[st.fieldInput, { height: 80, textAlignVertical: 'top', paddingTop: 12 }]} value={notes} onChangeText={setNotes} placeholder="Private notes — not shown to client" placeholderTextColor={C.textMute} multiline />
          </View>

          <TouchableOpacity style={[st.primaryBtn, { opacity: saving ? 0.7 : 1 }]} onPress={handleSave} disabled={saving} activeOpacity={0.8}>
            {saving ? <ActivityIndicator color={C.white} size="small" /> : <Text style={st.primaryBtnText}>Save & Generate Payment Link</Text>}
          </TouchableOpacity>

        </ScrollView>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

// ─── INVOICE DETAIL SCREEN ─────────────────────────────────────────────────
function InvoiceDetailScreen({ user, invoiceId, businessName, onBack, onNavigate }) {
  const [invoice, setInvoice] = useState(null);
  const [client, setClient] = useState(null);
  const [lineItems, setLineItems] = useState([]);
  const [loading, setLoading] = useState(true);
  const [updating, setUpdating] = useState(false);

  const load = async () => {
    setLoading(true);
    const [inv, items] = await Promise.all([
      supabase.getInvoice(invoiceId),
      supabase.getLineItems(invoiceId),
    ]);
    setInvoice(inv);
    setLineItems(items);
    // Load the client too, so their email is ready before any send.
    if (inv && inv.client_id) {
      const c = await supabase.getClient(inv.client_id);
      setClient(c);
    }
    setLoading(false);
  };

  useEffect(() => { if (invoiceId) load(); }, [invoiceId]);

  const handleMarkPaid = () => {
    showConfirm('Mark as Paid', 'Confirm this invoice has been paid in full?', async () => {
      setUpdating(true);
      const ok = await supabase.updateInvoice(invoice.id, { status: 'paid', paid_at: new Date().toISOString() });
      if (ok && invoice.client_id) {
        const client = await supabase.getClient(invoice.client_id);
        if (client?.email) {
          await supabase.sendReceipt(client.name || 'Client', client.email, invoice.total_due, invoice.id);
        }
      }
      setUpdating(false);
      if (ok) { load(); showAlert('Paid', 'Invoice marked as paid.'); }
      else showAlert('Error', 'Could not update invoice.');
    });
  };

  const handleShare = async () => {
    if (!invoice?.payment_link) { showAlert('No Link', 'No payment link available for this invoice.'); return; }
    // Use the client loaded with the invoice; fall back to a fresh fetch if needed.
    let c = client;
    if (!c && invoice?.client_id) { c = await supabase.getClient(invoice.client_id); }
    if (!c || !c.email) {
      showAlert('No email', 'This client has no email on file. Open the client and add an email address, then try again.');
      return;
    }
    setUpdating(true);
    const ok = await supabase.sendInvoice({
      clientName: c.name || 'Client',
      clientEmail: c.email,
      invoiceTitle: invoice.title || '',
      amount: invoice.total_due,
      invoiceId: invoice.id,
      paymentLink: invoice.payment_link || null,
      businessName: businessName || '',
    });
    setUpdating(false);
    if (ok) showAlert('Sent', 'Payment link sent to ' + c.email + '.');
    else showAlert('Could not send', 'The payment link could not be sent. Please try again.');
  };

  const handleOpenLink = () => {
    if (!invoice?.payment_link) { showAlert('No Link', 'No payment link available.'); return; }
    const { Linking } = require('react-native');
    Linking.openURL(invoice.payment_link);
  };

  const handleSendInvoice = async () => {
    // Use the client loaded with the invoice; fall back to a fresh fetch if needed.
    let c = client;
    if (!c && invoice?.client_id) { c = await supabase.getClient(invoice.client_id); }
    if (!c || !c.email) {
      showAlert('No email', 'This client has no email on file. Open the client and add an email address, then try again.');
      return;
    }
    setUpdating(true);
    const ok = await supabase.sendInvoice({
      clientName: c.name || 'Client',
      clientEmail: c.email,
      invoiceTitle: invoice.title || '',
      amount: invoice.total_due,
      invoiceId: invoice.id,
      paymentLink: invoice.payment_link || null,
      businessName: businessName || '',
    });
    setUpdating(false);
    if (ok) showAlert('Sent', 'Invoice sent to ' + c.email + '.');
    else showAlert('Could not send', 'The invoice could not be sent. Please try again.');
  };

  if (loading) return (
    <SafeAreaView style={st.safe}>
      <ScreenHeader title="Invoice" onBack={onBack} />
      <ActivityIndicator color={C.primary} style={{ marginTop: 60 }} />
    </SafeAreaView>
  );

  if (!invoice) return (
    <SafeAreaView style={st.safe}>
      <ScreenHeader title="Invoice" onBack={onBack} />
      <View style={{ alignItems: 'center', paddingTop: 60 }}><Text style={{ color: C.textMute }}>Invoice not found.</Text></View>
    </SafeAreaView>
  );

  const isPaid = invoice.status === 'paid';
  const canEdit = invoice.status === 'pending';

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor={C.background} />
      <ScreenHeader
        title="Invoice"
        onBack={onBack}
        rightAction={canEdit ? (
          <TouchableOpacity onPress={() => onNavigate('editInvoice', invoiceId)} style={{ paddingRight: 4 }}>
            <Text style={{ fontSize: 15, fontWeight: '600', color: C.primary }}>Edit</Text>
          </TouchableOpacity>
        ) : null}
      />
      <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={{ padding: H, paddingBottom: 40 }}>

        {/* HERO */}
        <View style={{ backgroundColor: C.primary, borderRadius: 16, padding: 20, marginBottom: 16, alignItems: 'center' }}>
          <Text style={{ fontSize: 11, color: C.primaryPale, fontWeight: '700', letterSpacing: 1, marginBottom: 6 }}>TOTAL DUE</Text>
          <Text style={{ fontSize: 38, fontWeight: '800', color: C.white, marginBottom: 10 }}>{fmt(invoice.total_due)}</Text>
          <Badge status={invoice.status} />
          {invoice.title ? <Text style={{ fontSize: 13, color: C.primaryPale, marginTop: 8, textAlign: 'center' }}>{invoice.title}</Text> : null}
        </View>

        {/* LINE ITEMS */}
        {lineItems.length > 0 && (
          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={[st.sectionLabel, { marginBottom: 12 }]}>LINE ITEMS</Text>
            {lineItems.map((item, idx) => (
              <View key={item.id} style={{ flexDirection: 'row', justifyContent: 'space-between', paddingVertical: 8, borderTopWidth: idx === 0 ? 0 : 1, borderTopColor: C.borderLight }}>
                <View style={{ flex: 1, marginRight: 8 }}>
                  <Text style={{ fontSize: 13, color: C.textDark, fontWeight: '600' }}>{item.description}</Text>
                  {item.quantity !== 1 && <Text style={{ fontSize: 11, color: C.textMute }}>Qty: {item.quantity}</Text>}
                </View>
                <Text style={{ fontSize: 13, fontWeight: '700', color: C.textDark }}>{fmt(item.amount)}</Text>
              </View>
            ))}
            <View style={{ borderTopWidth: 1, borderTopColor: C.border, paddingTop: 10, marginTop: 4 }}>
              {invoice.deposit_applied > 0 && (
                <View style={{ flexDirection: 'row', justifyContent: 'space-between', marginBottom: 6 }}>
                  <Text style={{ fontSize: 13, color: C.textMute }}>Deposit Applied</Text>
                  <Text style={{ fontSize: 13, color: C.paid }}>-{fmt(invoice.deposit_applied)}</Text>
                </View>
              )}
              <View style={{ flexDirection: 'row', justifyContent: 'space-between' }}>
                <Text style={{ fontSize: 15, fontWeight: '700', color: C.textDark }}>Total Due</Text>
                <Text style={{ fontSize: 15, fontWeight: '800', color: C.primary }}>{fmt(invoice.total_due)}</Text>
              </View>
            </View>
          </View>
        )}

        {/* DETAILS */}
        <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
          <Text style={[st.sectionLabel, { marginBottom: 12 }]}>DETAILS</Text>
          {[

// ─── EDIT PROPOSAL SCREEN ─────────────────────────────────────────────────
function EditProposalScreen({ user, proposalId, onBack, onSaved }) {
  const [title, setTitle] = useState('');
  const [description, setDescription] = useState('');
  const [amount, setAmount] = useState('');
  const [depositAmount, setDepositAmount] = useState('');
  const [validUntil, setValidUntil] = useState('');
  const [notes, setNotes] = useState('');
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    supabase.getProposal(proposalId).then(p => {
      if (p) {
        setTitle(p.title || '');
        setDescription(p.description || '');
        setAmount(p.amount ? String(p.amount) : '');
        setDepositAmount(p.deposit_amount ? String(p.deposit_amount) : '');
        setValidUntil(p.valid_until || '');
        setNotes(p.notes || '');
      }
      setLoading(false);
    });
  }, [proposalId]);

  const handleSave = async () => {
    if (!title.trim()) { showAlert('Required', 'Please enter a proposal title.'); return; }
    if (!amount || isNaN(parseFloat(amount))) { showAlert('Required', 'Please enter a valid amount.'); return; }
    setSaving(true);
    const updates = {
      title: title.trim(),
      description: description.trim() || null,
      amount: parseFloat(amount),
      deposit_amount: depositAmount ? parseFloat(depositAmount) : null,
      valid_until: validUntil || null,
      notes: notes.trim() || null,
    };
    const ok = await supabase.updateProposal(proposalId, updates);
    setSaving(false);
    if (ok) { onSaved(proposalId); }
    else { showAlert('Error', 'Could not save changes. Please try again.'); }
  };

  if (loading) return (
    <SafeAreaView style={st.safe}>
      <ScreenHeader title="Edit Proposal" onBack={onBack} />
      <ActivityIndicator color={C.primary} style={{ marginTop: 60 }} />
    </SafeAreaView>
  );

  return (

      <StatusBar barStyle="dark-content" backgroundColor={C.background} />
      <ScreenHeader
        title="Edit Proposal"
        onBack={onBack}
        rightAction={
          <TouchableOpacity onPress={handleSave} disabled={saving} style={{ paddingRight: 4 }}>
            {saving ? <ActivityIndicator color={C.primary} size="small" /> : <Text style={{ fontSize: 15, fontWeight: '700', color: C.primary }}>Save</Text>}
          </TouchableOpacity>
        }
      />
      <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
        <ScrollView contentContainerStyle={{ padding: H, paddingBottom: 60 }} keyboardShouldPersistTaps="handled" showsVerticalScrollIndicator={false}>
          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={st.sectionLabel}>PROPOSAL DETAILS</Text>
            <Text style={st.fieldLabel}>Title *</Text>
            <TextInput style={[st.fieldInput, { marginBottom: 14 }]} value={title} onChangeText={setTitle} placeholder="e.g. 3-Month Coaching Package" placeholderTextColor={C.textMute} autoCapitalize="words" />
            <Text style={st.fieldLabel}>Scope of Work</Text>
            <TextInput style={[st.fieldInput, { height: 100, textAlignVertical: 'top', paddingTop: 12 }]} value={description} onChangeText={setDescription} placeholder="Describe what's included..." placeholderTextColor={C.textMute} multiline />
          </View>

          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={st.sectionLabel}>PRICING</Text>
            <Text style={st.fieldLabel}>Total Amount *</Text>
            <View style={{ flexDirection: 'row', alignItems: 'center', marginBottom: 14 }}>
              <Text style={{ fontSize: 18, fontWeight: '600', color: C.textMid, marginRight: 8 }}>$</Text>
              <TextInput style={[st.fieldInput, { flex: 1 }]} value={amount} onChangeText={setAmount} placeholder="0.00" placeholderTextColor={C.textMute} keyboardType="decimal-pad" />
            </View>
            <Text style={st.fieldLabel}>Deposit Required</Text>
            <View style={{ flexDirection: 'row', alignItems: 'center', marginBottom: 14 }}>
              <Text style={{ fontSize: 18, fontWeight: '600', color: C.textMid, marginRight: 8 }}>$</Text>
              <TextInput style={[st.fieldInput, { flex: 1 }]} value={depositAmount} onChangeText={setDepositAmount} placeholder="Optional" placeholderTextColor={C.textMute} keyboardType="decimal-pad" />

          </View>

          <View style={[st.card, { padding: 16, marginBottom: 24 }]}>
            <Text style={st.sectionLabel}>INTERNAL NOTES</Text>

          </View>

          <TouchableOpacity style={[st.primaryBtn, { opacity: saving ? 0.7 : 1 }]} onPress={handleSave} disabled={saving} activeOpacity={0.8}>

// ─── NEW PROPOSAL SCREEN ───────────────────────────────────────────────────
function NewProposalScreen({ user, clientId, onBack, onSaved, onNavigate }) {
  const [clients, setClients] = useState([]);
  const [selectedClient, setSelectedClient] = useState(null);
  const [showClientPicker, setShowClientPicker] = useState(false);
  const [title, setTitle] = useState('');
  const [description, setDescription] = useState('');
  const [amount, setAmount] = useState('');
  const [depositAmount, setDepositAmount] = useState('');
  const [validUntil, setValidUntil] = useState('');
  const [notes, setNotes] = useState('');
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    supabase.getClients(user.id).then(data => {
      setClients(data);
      if (clientId) {
        const found = data.find(c => c.id === clientId);
        if (found) setSelectedClient(found);
      }
    });
  }, []);

  const handleSave = async (status = 'draft') => {
    if (!selectedClient) { showAlert('Required', 'Please select a client.'); return; }
    if (!title.trim()) { showAlert('Required', 'Please enter a proposal title.'); return; }
    if (!amount || isNaN(parseFloat(amount))) { showAlert('Required', 'Please enter a valid amount.'); return; }
    const limitCheck = await checkFreeLimit(user.id, user.subscriptionStatus || 'free', 'proposals');
    if (!limitCheck.allowed) { showAlert('Upgrade Required', limitCheck.message); return; }
    setSaving(true);
    const proposal = {
      user_id: user.id,
      client_id: selectedClient.id,
      title: title.trim(),
      description: description.trim() || null,
      amount: parseFloat(amount),
      deposit_amount: depositAmount ? parseFloat(depositAmount) : null,
      valid_until: validUntil || null,
      notes: notes.trim() || null,
      status,
      sent_at: status === 'sent' ? new Date().toISOString() : null,
    };
    const result = await supabase.addProposal(proposal);
    setSaving(false);
    if (result) {
      onSaved(result.id);
    } else {
      showAlert('Error', 'Could not save proposal. Please try again.');
    }
  };

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      <ScreenHeader
        title="New Proposal"
        onBack={onBack}
        rightAction={
          <TouchableOpacity onPress={() => handleSave('draft')} disabled={saving} style={{ paddingRight: 4 }}>

          </TouchableOpacity>

        <ScrollView contentContainerStyle={{ padding: H, paddingBottom: 60 }} keyboardShouldPersistTaps="handled" showsVerticalScrollIndicator={false}>

          {/* CLIENT PICKER */}
          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={st.sectionLabel}>CLIENT</Text>
            <TouchableOpacity
              style={[st.fieldInput, { flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between' }]}
              onPress={() => setShowClientPicker(!showClientPicker)}
              activeOpacity={0.7}
            >
              <Text style={{ fontSize: 15, color: selectedClient ? C.textDark : C.textMute }}>
                {selectedClient ? selectedClient.name : 'Select a client...'}
              </Text>
              <Ionicons name={showClientPicker ? 'chevron-up' : 'chevron-down'} size={18} color={C.textMute} />
            </TouchableOpacity>
            {showClientPicker && (
              <View style={{ marginTop: 8, borderWidth: 1, borderColor: C.border, borderRadius: 10, overflow: 'hidden' }}>
                {clients.length === 0 ? (
                  <TouchableOpacity style={{ padding: 14 }} onPress={() => onNavigate('newClient')}>

              style={[st.fieldInput, { height: 100, textAlignVertical: 'top', paddingTop: 12 }]}

              placeholderTextColor={C.textMute}

            />
          </View>

          {/* PRICING */}
          <View style={[st.card, { padding: 16, marginBottom: 12 }]}>
            <Text style={st.sectionLabel}>PRICING</Text>
            <Text style={st.fieldLabel}>Total Amount *</Text>
            <View style={{ flexDirection: 'row', alignItems: 'center' }}>
              <Text style={{ fontSize: 18, fontWeight: '600', color: C.textMid, marginRight: 8 }}>$</Text>
              <TextInput

                value={amount}

                placeholderTextColor={C.textMute}
                keyboardType="decimal-pad"
              />

            <View style={{ flexDirection: 'row', alignItems: 'center' }}>

// ─── PROPOSAL DETAIL SCREEN ────────────────────────────────────────────────
function ProposalDetailScreen({ user, proposalId, onBack, onNavigate }) {
  const [proposal, setProposal] = useState(null);
  const [loading, setLoading] = useState(true);
  const [updating, setUpdating] = useState(false);

  const load = async () => {
    setLoading(true);
    const p = await supabase.getProposal(proposalId);
    setProposal(p);
    setLoading(false);
  };

  useEffect(() => { if (proposalId) load(); }, [proposalId]);

  const handleUpdate = async (updates, successMsg) => {
    setUpdating(true);
    const ok = await supabase.updateProposal(proposal.id, updates);
    setUpdating(false);
    if (ok) {
      showAlert('Updated', successMsg);
      load();
    } else {
      showAlert('Error', 'Could not update proposal. Please try again.');
    }
  };

  if (loading) return (
    <SafeAreaView style={st.safe}>
      <ScreenHeader title="Proposal" onBack={onBack} />
      <ActivityIndicator color={C.primary} style={{ marginTop: 60 }} />
    </SafeAreaView>
  );

  if (!proposal) return (
    <SafeAreaView style={st.safe}>
      <ScreenHeader title="Proposal" onBack={onBack} />
      <View style={{ alignItems: 'center', paddingTop: 60 }}>
        <Text style={{ color: C.textMute }}>Proposal not found.</Text>
      </View>
    </SafeAreaView>
  );

  const canEdit = proposal.status === 'draft';

  const statusActions = () => {
    if (proposal.status === 'draft') return (
      <TouchableOpacity style={[st.primaryBtn, { marginBottom: 10 }]} onPress={() => handleUpdate({ status: 'sent', sent_at: new Date().toISOString() }, 'Proposal marked as sent.')} disabled={updating} activeOpacity={0.8}>
        <Text style={st.primaryBtnText}>Mark as Sent</Text>
      </TouchableOpacity>
    );
    if (proposal.status === 'sent') return (
      <View style={{ gap: 10 }}>
        <TouchableOpacity style={[st.primaryBtn]} onPress={() => handleUpdate({ status: 'accepted', accepted_at: new Date().toISOString() }, 'Proposal accepted!') } disabled={updating} activeOpacity={0.8}>
          <Text style={st.primaryBtnText}>Mark as Accepted</Text>
        </TouchableOpacity>
        <TouchableOpacity style={[st.outlineBtn, { borderColor: C.overdue }]} onPress={() => handleUpdate({ status: 'declined' }, 'Proposal marked as declined.')} disabled={updating} activeOpacity={0.8}>
          <Text style={[st.outlineBtnText, { color: C.overdue }]}>Mark as Declined</Text>
        </TouchableOpacity>
      </View>
    );
    if (proposal.status === 'accepted') return (
      <View style={{ gap: 10 }}>
        <TouchableOpacity style={[st.primaryBtn]} onPress={() => onNavigate('newContract', { clientId: proposal.client_id, title: proposal.title, amount: proposal.amount, deposit: proposal.deposit_amount })} activeOpacity={0.8}>
          <Text style={st.primaryBtnText}>Create Contract from This</Text>
        </TouchableOpacity>
        <TouchableOpacity style={[st.outlineBtn, { borderColor: C.primary }]} onPress={() => onNavigate('newInvoice', { clientId: proposal.client_id, title: proposal.title, amount: proposal.amount })} activeOpacity={0.8}>
          <Text style={[st.outlineBtnText, { color: C.primary }]}>Create Invoice from This</Text>
        </TouchableOpacity>
      </View>
    );
    return null;
  };

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      <ScreenHeader
        title="Proposal"
        onBack={onBack}
        rightAction={canEdit ? (
          <TouchableOpacity onPress={() => onNavigate('editProposal', proposalId)} style={{ paddingRight: 4 }}>
            <Text style={{ fontSize: 15, fontWeight: '600', color: C.primary }}>Edit</Text>
          </TouchableOpacity>
        ) : null}
      />
      <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={{ padding: H, paddingBottom: 40 }}>

        {/* HERO */}

function NewExpenseScreen({ user, onBack, onSaved, editItem }) {
  const isEdit = !!editItem;
  const [vendor, setVendor] = useState(editItem ? (editItem.vendor || '') : '');
  const [amount, setAmount] = useState(editItem ? String(editItem.amount) : '');
  const [category, setCategory] = useState(editItem ? (editItem.category || 'Other') : 'Other');
  const [expenseDate, setExpenseDate] = useState(editItem ? editItem.expense_date : new Date().toISOString().slice(0, 10));
  const [notes, setNotes] = useState(editItem ? (editItem.notes || '') : '');
  const [saving, setSaving] = useState(false);

  const handleSave = async () => {
    const amt = parseFloat(amount);
    if (!vendor.trim()) { showAlert('Add a vendor', 'Who did you pay? (e.g. Adobe, Home Depot)'); return; }
    if (!amt || amt <= 0) { showAlert('Add an amount', 'Enter how much you spent.'); return; }
    setSaving(true);
    const fields = {
      vendor: vendor.trim(),
      category: category,
      amount: amt,
      expense_date: expenseDate,
      notes: notes.trim() || null,
    };
    let ok;
    if (isEdit) {
      ok = await supabase.updateExpense(editItem.id, fields);
    } else {
      ok = await supabase.addExpense({ id: 'exp_' + Date.now().toString(36) + Math.random().toString(36).slice(2, 7), user_id: user.id, ...fields });
    }
    setSaving(false);
    if (ok) { onSaved(); } else { showAlert('Could not save', 'Please try again.'); }
  };

  return (
    <SafeAreaView style={st.safe}>
      <ScreenHeader title={isEdit ? "Edit Expense" : "New Expense"} onBack={onBack} />
      <ScrollView contentContainerStyle={{ paddingHorizontal: H, paddingBottom: 60 }} showsVerticalScrollIndicator={false} keyboardShouldPersistTaps="handled">
        <Text style={st.fieldLabel}>Vendor</Text>
        <TextInput style={st.fieldInput} value={vendor} onChangeText={setVendor} placeholder="Who did you pay?" placeholderTextColor={C.textMute} />

        <Text style={st.fieldLabel}>Amount</Text>
        <TextInput style={st.fieldInput} value={amount} onChangeText={setAmount} placeholder="0.00" placeholderTextColor={C.textMute} keyboardType="decimal-pad" />

        <Text style={st.fieldLabel}>Category</Text>
        <View style={{ flexDirection: 'row', flexWrap: 'wrap', gap: 8, marginBottom: 16 }}>
          {EXPENSE_CATEGORIES.map((cat) => (
            <TouchableOpacity
              key={cat}
              style={{ paddingVertical: 8, paddingHorizontal: 16, borderRadius: 20, borderWidth: 1.5, borderColor: category === cat ? C.primary : C.border, backgroundColor: category === cat ? C.primary : 'transparent' }}
              onPress={() => setCategory(cat)}
              activeOpacity={0.7}
            >
              <Text style={{ fontSize: 13, fontWeight: '600', color: category === cat ? C.white : C.textMute }}>{cat}</Text>
            </TouchableOpacity>
          ))}
        </View>

        <Text style={st.fieldLabel}>Date</Text>
        <TextInput style={st.fieldInput} value={expenseDate} onChangeText={setExpenseDate} placeholder="YYYY-MM-DD" placeholderTextColor={C.textMute} />

        <Text style={st.fieldLabel}>Notes (optional)</Text>
        <TextInput style={[st.fieldInput, { height: 80, textAlignVertical: 'top' }]} value={notes} onChangeText={setNotes} placeholder="Anything to remember about this expense" placeholderTextColor={C.textMute} multiline />

        <TouchableOpacity style={[st.primaryBtn, { marginTop: 24 }]} onPress={handleSave} disabled={saving} activeOpacity={0.85}>
          <Text style={st.primaryBtnText}>{saving ? 'Saving...' : (isEdit ? 'Update Expense' : 'Save Expense')}</Text>
        </TouchableOpacity>
      </ScrollView>
    </SafeAreaView>
  );
}

// ─── APPOINTMENTS ──────────────────────────────────────────────────────────
function AppointmentsListScreen({ user, onNavigate, onBack }) {
  const [appts, setAppts] = useState([]);
  const [clients, setClients] = useState({});
  const [loading, setLoading] = useState(true);

  const load = async () => {
    setLoading(true);
    const data = await supabase.getAppointments(user.id);
    const clientList = await supabase.getClients(user.id);
    const map = {};
    clientList.forEach((c) => { map[c.id] = c.name; });
    setClients(map);
    setAppts(data);
    setLoading(false);
  };

  useEffect(() => { load(); }, []);

  const fmtDate = (d) => d ? new Date(d + 'T00:00:00').toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' }) : '';
  const fmtTime = (t) => {
    if (!t) return '';
    const parts = t.split(':');
    let h = parseInt(parts[0], 10);
    const m = parts[1];
    const ampm = h >= 12 ? 'PM' : 'AM';
    h = h % 12; if (h === 0) h = 12;
    return h + ':' + m + ' ' + ampm;
  };

  const handleDelete = (appt) => {
    showConfirm('Delete appointment?', 'This will remove it from your schedule.', async () => {
      const ok = await supabase.deleteAppointment(appt.id);
      if (ok) { load(); } else { showAlert('Could not delete', 'Please try again.'); }
    });
  };

  const todayStr = new Date().toISOString().slice(0, 10);
  const upcoming = appts.filter((a) => a.appointment_date >= todayStr);
  const past = appts.filter((a) => a.appointment_date < todayStr);

  const renderAppt = (appt) => (
    <TouchableOpacity key={appt.id} style={[st.card, { flexDirection: 'row', alignItems: 'center', padding: 16, marginBottom: 10 }]} onPress={() => onNavigate('newAppointment', appt)} onLongPress={() => handleDelete(appt)} activeOpacity={0.7}>
      <View style={{ width: 52, alignItems: 'center', marginRight: 14 }}>
        <Text style={{ fontSize: 11, fontWeight: '700', color: C.primary, textTransform: 'uppercase' }}>{fmtDate(appt.appointment_date).split(' ')[0]}</Text>
        <Text style={{ fontSize: 20, fontWeight: '800', color: C.textDark }}>{new Date(appt.appointment_date + 'T00:00:00').getDate()}</Text>
      </View>
      <View style={{ flex: 1 }}>
        <Text style={{ fontSize: 15, fontWeight: '700', color: C.textDark, marginBottom: 2 }}>{appt.title}</Text>
        <Text style={{ fontSize: 12, color: C.textMute }}>
          {appt.start_time ? fmtTime(appt.start_time) : 'All day'}
          {appt.client_id && clients[appt.client_id] ? ' · ' + clients[appt.client_id] : ''}
          {appt.location ? ' · ' + appt.location : ''}
        </Text>
      </View>
    </TouchableOpacity>
  );

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      <View style={{ flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between', paddingHorizontal: H, paddingTop: 20, paddingBottom: 16 }}>
        <View>

// ─── TIME TRACKING ─────────────────────────────────────────────────────────
function TimeListScreen({ user, onNavigate, onBack }) {
  const [entries, setEntries] = useState([]);
  const [clients, setClients] = useState({});
  const [loading, setLoading] = useState(true);

  const load = async () => {
    setLoading(true);
    const data = await supabase.getTimeEntries(user.id);
    const clientList = await supabase.getClients(user.id);
    const map = {};
    clientList.forEach((c) => { map[c.id] = c.name; });
    setClients(map);
    setEntries(data);
    setLoading(false);
  };

  useEffect(() => { load(); }, []);

  const totalHours = entries.reduce((sum, e) => sum + (parseFloat(e.hours) || 0), 0);
  const totalValue = entries.reduce((sum, e) => sum + ((parseFloat(e.hours) || 0) * (parseFloat(e.hourly_rate) || 0)), 0);

  // Group billable entries (have a client AND a rate) by client for one-tap invoicing.
  const billableByClient = (function () {
    const groups = {};
    entries.forEach((e) => {
      if (e.client_id && e.hourly_rate && parseFloat(e.hourly_rate) > 0) {
        if (!groups[e.client_id]) groups[e.client_id] = { clientId: e.client_id, clientName: clients[e.client_id] || 'Client', hours: 0, entries: [] };
        groups[e.client_id].hours += parseFloat(e.hours) || 0;
        groups[e.client_id].entries.push(e);
      }
    });
    return Object.values(groups);
  })();
  const fmtMoney = (n) => '$' + Number(n || 0).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  const fmtDate = (d) => d ? new Date(d + 'T00:00:00').toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' }) : '';

  const handleDelete = (entry) => {
    showConfirm('Delete time entry?', 'This will remove it from your log.', async () => {
      const ok = await supabase.deleteTimeEntry(entry.id);
      if (ok) { load(); } else { showAlert('Could not delete', 'Please try again.'); }
    });
  };

  return (
    <SafeAreaView style={st.safe}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      <View style={{ flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between', paddingHorizontal: H, paddingTop: 20, paddingBottom: 16 }}>
        <View>
          <Text style={{ fontSize: 26, fontWeight: '800', color: C.textDark, marginBottom: 4 }}>Time</Text>
          <Text style={{ fontSize: 14, color: C.textMute }}>Track your billable hours</Text>
        </View>
        <TouchableOpacity style={st.addBtn} onPress={() => onNavigate('newTime')} activeOpacity={0.8}>
          <Ionicons name="add" size={24} color={C.white} />
        </TouchableOpacity>
      </View>

      <View style={[st.card, { marginHorizontal: H, marginBottom: 12, padding: 20, backgroundColor: C.primary, flexDirection: 'row', justifyContent: 'space-between' }]}>
        <View>
          <Text style={{ fontSize: 13, color: C.white, opacity: 0.85, marginBottom: 4 }}>Total hours</Text>
          <Text style={{ fontSize: 28, fontWeight: '800', color: C.white }}>{totalHours.toFixed(1)}</Text>
        </View>
        <View style={{ alignItems: 'flex-end' }}>
          <Text style={{ fontSize: 13, color: C.white, opacity: 0.85, marginBottom: 4 }}>Value</Text>
          <Text style={{ fontSize: 28, fontWeight: '800', color: C.white }}>{fmtMoney(totalValue)}</Text>
        </View>
      </View>

      {billableByClient.length > 0 && (
        <View style={{ marginHorizontal: H, marginBottom: 16 }}>
          {billableByClient.map((grp) => (
            <TouchableOpacity
              key={grp.clientId}
              style={[st.outlineBtn, { borderColor: C.primary, marginBottom: 8, flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', paddingHorizontal: 16 }]}
              activeOpacity={0.8}
              onPress={() => onNavigate('newInvoice', {
                clientId: grp.clientId,
                title: 'Hours — ' + grp.clientName,
                timeEntries: grp.entries,
              })}
            >
              <Text style={[st.outlineBtnText, { color: C.primary }]}>Bill {grp.hours.toFixed(1)} hrs to {grp.clientName}</Text>
              <Ionicons name="arrow-forward" size={16} color={C.primary} />
            </TouchableOpacity>
          ))}
        </View>
      )}

      <ScrollView contentContainerStyle={{ paddingHorizontal: H, paddingBottom: 40 }} showsVerticalScrollIndicator={false}>
        {loading ? (
          <Text style={{ textAlign: 'center', color: C.textMute, marginTop: 40 }}>Loading...</Text>
        ) : entries.length === 0 ? (
          <View style={{ alignItems: 'center', marginTop: 40 }}>
            <Ionicons name="time-outline" size={52} color={C.border} style={{ marginBottom: 16 }} />
            <Text style={{ fontSize: 16, fontWeight: '700', color: C.textMute, marginBottom: 6 }}>No time logged yet</Text>
            <Text style={{ fontSize: 14, color: C.textMute, textAlign: 'center', marginBottom: 20 }}>Log your first block of work to start tracking.</Text>
            <TouchableOpacity style={[st.primaryBtn, { paddingHorizontal: 32 }]} onPress={() => onNavigate('newTime')}>
              <Text style={st.primaryBtnText}>Log Time</Text>
            </TouchableOpacity>
          </View>
        ) : (
          entries.map((e) => (
            <TouchableOpacity key={e.id} style={[st.card, { flexDirection: 'row', alignItems: 'center', padding: 16, marginBottom: 10 }]} onPress={() => onNavigate('newTime', e)} onLongPress={() => handleDelete(e)} activeOpacity={0.7}>
              <View style={{ width: 44, height: 44, borderRadius: 12, backgroundColor: C.cardAlt, alignItems: 'center', justifyContent: 'center', marginRight: 14 }}>
                <Ionicons name="time-outline" size={22} color={C.primary} />
              </View>
              <View style={{ flex: 1 }}>
                <Text style={{ fontSize: 15, fontWeight: '700', color: C.textDark, marginBottom: 2 }}>{e.description}</Text>
                <Text style={{ fontSize: 12, color: C.textMute }}>
                  {parseFloat(e.hours).toFixed(1)} hr{e.client_id && clients[e.client_id] ? ' · ' + clients[e.client_id] : ''} · {fmtDate(e.entry_date)}
                </Text>
              </View>
              {e.hourly_rate ? (
                <Text style={{ fontSize: 15, fontWeight: '800', color: C.textDark }}>{fmtMoney(parseFloat(e.hours) * parseFloat(e.hourly_rate))}</Text>
              ) : null}
            </TouchableOpacity>
          ))
        )}
        {entries.length > 0 && (
          <Text style={{ fontSize: 12, color: C.textMute, textAlign: 'center', marginTop: 12 }}>Tip: press and hold an entry to delete it.</Text>
        )}
      </ScrollView>
    </SafeAreaView>
  );
}

// ─── MAIN APP ──────────────────────────────────────────────────────────────
export default function App() {
  const [user, setUser] = useState(null);
  const [authScreen, setAuthScreen] = useState('login');
  const [screen, setScreen] = useState('home');
  const [screenParam, setScreenParam] = useState(null);
  const [prevScreen, setPrevScreen] = useState('home');
  const [profileName, setProfileName] = useState('');
  const [businessName, setBusinessName] = useState('');
  const [logoUrl, setLogoUrl] = useState(null);
  const [subscriptionStatus, setSubscriptionStatus] = useState('free');
  const [appReady, setAppReady] = useState(false);
  const [showOnboarding, setShowOnboarding] = useState(false);

  // AUTO-LOGIN
  useEffect(() => {
    const tryAutoLogin = async () => {
      try {
        const saved = await AsyncStorage.getItem('bp_savedSession');
        if (saved) {
          const { refresh_token } = JSON.parse(saved);
          const result = await supabase.refreshSession(refresh_token);
          if (result.user && !result.error) {
            const isDeactivated = await supabase.checkDeletion(result.user.id);
            if (!isDeactivated) {
              await initUser(result.user, result.session);
              await AsyncStorage.setItem('bp_savedSession', JSON.stringify({
                user: result.user,
                refresh_token: result.session.refresh_token,
              }));
            } else {
              await AsyncStorage.removeItem('bp_savedSession');
            }
          } else {
            await AsyncStorage.removeItem('bp_savedSession');
          }
        }
      } catch (e) {
        await AsyncStorage.removeItem('bp_savedSession');
      }
      setAppReady(true);
    };
    tryAutoLogin();
  }, []);

  const initUser = async (loggedInUser, session) => {
    setUser(loggedInUser);
    // Store session token for authenticated Supabase requests
    if (session?.access_token) setSessionToken(session.access_token);
    else if (loggedInUser?.access_token) setSessionToken(loggedInUser.access_token);
    const fullName = loggedInUser?.user_metadata?.full_name || '';
    const firstName = fullName ? fullName.trim().split(' ')[0] : (loggedInUser?.email || 'there');
    setProfileName(firstName);
    const profile = await supabase.getProfile(loggedInUser.id);
    if (profile?.business_name) setBusinessName(profile.business_name);
    if (profile?.logo_url) setLogoUrl(profile.logo_url);
    const bizhqPlan = profile?.bp_subscription_status === 'pro' ? 'pro' : 'free';
    setSubscriptionStatus(bizhqPlan);
    setPlanStatus(bizhqPlan);
    try {
      const seen = await AsyncStorage.getItem('bp_onboardingComplete');
      if (!seen) setShowOnboarding(true);
    } catch (e) {}
  };

  const handleLogin = async (loggedInUser, session) => {
    const isDeactivated = await supabase.checkDeletion(loggedInUser.id);
    if (isDeactivated) {
      showAlert('Account Deactivated', 'This account has been deactivated. Contact support@invoicespro.org to recover it.');
      return;
    }
    await initUser(loggedInUser, session);
  };

  const handleSignOut = async () => {
    try { await AsyncStorage.removeItem('bp_savedSession'); } catch (e) {}
    setSessionToken(null);
    setUser(null);
    setAuthScreen('login');
    setScreen('home');
    setProfileName('');
    setBusinessName('');
    setSubscriptionStatus('free');
    setPlanStatus('free');
  };

  const handleDeleteAccount = () => {
    showConfirm(
      'Delete Account',
      'Your account will be deactivated immediately. All data will be permanently deleted after 30 days. You can recover your account within that window by contacting support@invoicespro.org.\n\nAre you sure?',
      async () => {
        const success = await supabase.requestDeletion(user.id);
        if (success) {
          showAlert('Account Deactivated', 'Your account has been deactivated. Contact support@invoicespro.org within 30 days if you wish to recover it.');
          handleSignOut();
        } else {
          showAlert('Error', 'Could not process your request. Please try again.');
        }
      }
    );
  };

  const handleUpgrade = async (interval = 'month') => {
    if (!user) return;
    try {
      const res = await fetch(SUBSCRIPTION_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ action: 'checkout', app: 'bizhq', interval, userId: user.id, email: user.email }),
      });
      const data = await res.json();
      if (data.url) {
        const { Linking } = require('react-native');
        Linking.openURL(data.url);
      } else {
        showAlert('Error', data.error || 'Could not start checkout.');
      }
    } catch (e) {
      showAlert('Error', 'Could not connect to payment service.');
    }
  };

  const handleManageSubscription = async () => {
    if (!user) return;
    try {
      const res = await fetch(SUBSCRIPTION_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ action: 'portal', userId: user.id, email: user.email }),
      });
      const data = await res.json();
      if (data.url) {
        const { Linking } = require('react-native');
        Linking.openURL(data.url);
      }
    } catch (e) {
      showAlert('Error', 'Could not open subscription management.');
    }
  };

  const navigate = (dest, param = null) => {
    setPrevScreen(screen);
    setScreenParam(param);
    setScreen(dest);
  };

  // ── LOADING ──
  if (!appReady) {
    return (
      <SafeAreaView style={[st.safe, { justifyContent: 'center', alignItems: 'center' }]}>
        <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
        <View style={{ flexDirection: 'row', alignItems: 'baseline', marginBottom: 20 }}>
          <Text style={[st.appName, { color: C.textDark }]}>Biz</Text>
          <Text style={[st.appNameAccent, { color: C.primary }]}>HQ</Text>
        </View>
        <ActivityIndicator color={C.primary} size="small" />
      </SafeAreaView>
    );
  }

  // ── AUTH ──
  if (!user) {
    if (authScreen === 'signup') return <SignUpScreen onSignUp={handleLogin} onGoLogin={() => setAuthScreen('login')} />;
    return <LoginScreen onLogin={handleLogin} onGoSignUp={() => setAuthScreen('signup')} />;
  }

  // ── ONBOARDING ──
  if (showOnboarding) return <OnboardingScreen user={user} onFinish={() => setShowOnboarding(false)} />;

  // ── SCREENS ──
  const renderScreen = () => {
    switch (screen) {
      case 'home':
        return <HomeScreen user={user} profileName={profileName} businessName={businessName} logoUrl={logoUrl} onNavigate={navigate} />;
      case 'clients':
        return <ClientsScreen user={user} onNavigate={navigate} />;
      case 'create':
        return <CreateScreen onNavigate={navigate} />;
      case 'settings':
        return <SettingsScreen user={user} profileName={profileName} businessName={businessName} logoUrl={logoUrl} subscriptionStatus={subscriptionStatus} onSignOut={handleSignOut} onNavigate={navigate} onDeleteAccount={handleDeleteAccount} onUpgrade={handleUpgrade} onManageSubscription={handleManageSubscription} />;
      // Placeholder screens — will be built next
      case 'newClient':      return <NewClientScreen user={user} onBack={() => navigate('clients')} onSaved={() => navigate('clients')} />;
      case 'clientDetail':   return <ClientDetailScreen user={user} clientId={screenParam} onBack={() => navigate('clients')} onNavigate={navigate} />;
      case 'editClient':     return <EditClientScreen user={user} clientId={screenParam} onBack={() => navigate('clientDetail', screenParam)} onSaved={(id) => navigate('clientDetail', id)} />;
      case 'newProposal':    return <NewProposalScreen user={user} clientId={screenParam} onBack={() => navigate(screenParam ? 'clientDetail' : 'create')} onSaved={(id) => navigate('proposalDetail', id)} onNavigate={navigate} />;
      case 'proposalDetail': return <ProposalDetailScreen user={user} proposalId={screenParam} onBack={() => navigate(prevScreen || 'proposals')} onNavigate={navigate} />;
      case 'editProposal':   return <EditProposalScreen user={user} proposalId={screenParam} onBack={() => navigate('proposalDetail', screenParam)} onSaved={(id) => navigate('proposalDetail', id)} />;
      case 'newContract':    return <NewContractScreen user={user} clientId={screenParam} onBack={() => navigate(screenParam ? 'clientDetail' : 'create')} onSaved={(id) => navigate('contractDetail', id)} onNavigate={navigate} />;
      case 'contractDetail': return <ContractDetailScreen user={user} contractId={screenParam} onBack={() => navigate(prevScreen || 'contracts')} onNavigate={navigate} />;
      case 'newInvoice':     return <NewInvoiceScreen user={user} clientId={screenParam} onBack={() => navigate(screenParam ? 'clientDetail' : 'create')} onSaved={(id) => navigate('invoiceDetail', id)} onNavigate={navigate} />;
      case 'invoiceDetail':  return <InvoiceDetailScreen user={user} invoiceId={screenParam} businessName={businessName} onBack={() => navigate(prevScreen || 'invoices')} onNavigate={navigate} />;
      case 'editInvoice':    return <EditInvoiceScreen user={user} invoiceId={screenParam} onBack={() => navigate('invoiceDetail', screenParam)} onSaved={(id) => navigate('invoiceDetail', id)} />;
      case 'proposals':      return <ProposalsListScreen user={user} onNavigate={navigate} onBack={() => navigate('home')} />;
      case 'contracts':      return <ContractsListScreen user={user} onNavigate={navigate} onBack={() => navigate('home')} />;
      case 'invoices':       return <InvoicesListScreen user={user} onNavigate={navigate} onBack={() => navigate('home')} />;
      case 'commandBar':     return <CommandBarScreen user={user} businessName={businessName} onBack={() => navigate('home')} onNavigate={navigate} />;
      case 'archived':       return <ArchivedScreen user={user} onBack={() => navigate('create')} onNavigate={navigate} />;
      case 'editProfile':    return <EditProfileScreen user={user} profileName={profileName} businessName={businessName} logoUrl={logoUrl} onBack={() => navigate('settings')} onSaved={(name, biz, logo) => { setProfileName(name); setBusinessName(biz); if (logo) setLogoUrl(logo); navigate('settings'); }} />;
      case 'documents':      return <DocumentsScreen user={user} onBack={() => navigate('create')} onNavigate={navigate} />;
      case 'expenses':       return <ExpensesListScreen user={user} onNavigate={navigate} onBack={() => navigate('create')} />;
      case 'newExpense':     return <NewExpenseScreen user={user} editItem={(screenParam && typeof screenParam === 'object') ? screenParam : null} onBack={() => navigate('expenses')} onSaved={() => navigate('expenses')} />;
      case 'appointments':   return <AppointmentsListScreen user={user} onNavigate={navigate} onBack={() => navigate('create')} />;
      case 'newAppointment': return <NewAppointmentScreen user={user} editItem={(screenParam && typeof screenParam === 'object') ? screenParam : null} onBack={() => navigate('appointments')} onSaved={() => navigate('appointments')} />;
      case 'time':           return <TimeListScreen user={user} onNavigate={navigate} onBack={() => navigate('create')} />;
      case 'newTime':        return <NewTimeScreen user={user} editItem={(screenParam && typeof screenParam === 'object') ? screenParam : null} onBack={() => navigate('time')} onSaved={() => navigate('time')} />;
      case 'privacy':        return <PrivacyPolicyScreen onBack={() => navigate('settings')} />;
      case 'terms':          return <TermsOfServiceScreen onBack={() => navigate('settings')} />;
      default:
        return <HomeScreen user={user} profileName={profileName} businessName={businessName} logoUrl={logoUrl} onNavigate={navigate} />;
    }
  };

  // BOTTOM NAV TABS
  const navTabs = [
    { key: 'home',    label: 'Home',    icon: 'home-outline',    activeIcon: 'home' },
    { key: 'clients', label: 'Clients', icon: 'people-outline',  activeIcon: 'people' },
    { key: 'create',  label: '',        icon: 'add',             activeIcon: 'add', isFab: true },
    { key: 'commandBar', label: 'HQ AI', icon: 'chatbubble-ellipses-outline', activeIcon: 'chatbubble-ellipses' },
    { key: 'settings',label: 'Settings',icon: 'settings-outline',activeIcon: 'settings' },
  ];

  const activeTab = ['home','clients','create','commandBar','settings'].includes(screen) ? screen : 'home';

  return (
    <View style={{ flex: 1, backgroundColor: C.background }}>
      <StatusBar barStyle="dark-content" backgroundColor="#7A4A30" />
      {renderScreen()}
      <View style={st.bottomNav}>
        {navTabs.map(tab => {
          const isActive = activeTab === tab.key;
          if (tab.isFab) {
            return (
              <TouchableOpacity
                key={tab.key}
                style={{ flex: 1, alignItems: 'center', justifyContent: 'center' }}
                onPress={() => navigate(tab.key)}
                activeOpacity={0.8}
              >
                <View style={{
                  width: 48, height: 48, borderRadius: 24,
                  backgroundColor: C.primary,
                  alignItems: 'center', justifyContent: 'center',
                  marginBottom: 4,
                  shadowColor: C.primary, shadowOffset: { width: 0, height: 4 },
                  shadowOpacity: 0.4, shadowRadius: 8, elevation: 8,
                }}>
                  <Ionicons name="add" size={28} color={C.white} />
                </View>
              </TouchableOpacity>
            );
          }
          return (
            <TouchableOpacity key={tab.key} style={st.navItem} onPress={() => navigate(tab.key)} activeOpacity={0.7}>
              <Ionicons
                name={isActive ? tab.activeIcon : tab.icon}
                size={22}
                color={isActive ? C.primary : C.textMute}
                style={{ marginBottom: 3 }}
              />
              <Text style={[st.navLabel, isActive && { color: C.primary, fontWeight: '700' }]}>{tab.label}</Text>
              {isActive && <View style={{ width: 4, height: 4, borderRadius: 2, backgroundColor: C.primary, marginTop: 2 }} />}
            </TouchableOpacity>
          );
        })}
      </View>
    </View>
  );
}
